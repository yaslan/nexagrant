// setVisibilityAction'ı tarayıcının göndereceği POST'un aynısıyla çağırır (Next-Action başlığı + encodeReply gövdesi).
// Token'lar next-auth/jwt encode() ile, CI'nın test AUTH_SECRET fikstürüyle üretilir; çerez basılmaz.
import { execFileSync } from 'node:child_process';
import { encode } from 'next-auth/jwt';
const B = 'http://127.0.0.1:47931';
const [ACTION, ADMIN_ID, MEMBER_ID] = process.argv.slice(2);
const psql = (sql) => execFileSync('docker', ['exec', 'csfb-acb-pg', 'psql', '-U', 'fabrika', '-d', 'fabrika', '-tAc', sql]).toString().trim();
const token = (id) => encode({ token: { id, sub: id }, secret: process.env.AUTH_SECRET, salt: 'authjs.session-token', maxAge: 3600 });
const vis = () => psql(`select visibility from "Plan" where id='ultra'`);
const audits = () => psql(`select count(*) from "AuditEvent" where type='plan.visibility.changed' and target='ultra'`);
async function call(cookie, args) {
  const res = await fetch(`${B}/tr/admin/billing/plans`, {
    method: 'POST',
    headers: { 'Next-Action': ACTION, Accept: 'text/x-component', 'Content-Type': 'text/plain;charset=UTF-8', Origin: B, Cookie: `authjs.session-token=${cookie}` },
    body: JSON.stringify(args),
  });
  const body = await res.text();
  return { status: res.status, result: (body.match(/^1:(.*)$/m) || [])[1] ?? '<yok> ' + body.slice(0, 200) };
}
const admin = await token(ADMIN_ID);
const member = await token(MEMBER_ID);
const rows = [];
const run = async (label, cookie, args, expect) => {
  const before = vis(), a0 = audits();
  const r = await call(cookie, args);
  const after = vis(), a1 = audits();
  const ok = r.status === 200 && r.result === expect;
  console.log(`[${label}] args=${JSON.stringify(args).slice(0, 60)} HTTP ${r.status} sonuç=${r.result} | Plan.visibility ${before}→${after} | audit ${a0}→${a1} | ${ok ? 'BEKLENEN' : 'BEKLENMEYEN'}`);
  rows.push(ok);
  return { before, after };
};
for (const bad of ['csfb-acb', 'PUBLIC', '', 'public ', 'archived\u0000']) {
  const { before, after } = await run(`admin, geçersiz ${JSON.stringify(bad)}`, admin, ['ultra', bad], '{"ok":false,"code":"INVALID_VISIBILITY"}');
  if (before !== after) rows.push(false);
}
for (const bad of [null, 42, ['public'], { v: 'public' }]) {
  await run(`admin, dize olmayan ${JSON.stringify(bad)}`, admin, ['ultra', bad], '{"ok":false,"code":"INVALID_VISIBILITY"}');
}
await run('üye (admin değil), geçersiz', member, ['ultra', 'csfb-acb'], '{"ok":false,"code":"PLATFORM_ADMIN_REQUIRED"}');
const p = await run('admin, geçerli private', admin, ['ultra', 'private'], '{"ok":true}');
if (p.after !== 'private') rows.push(false);
const r = await run('admin, geri public', admin, ['ultra', 'public'], '{"ok":true}');
if (r.after !== 'public') rows.push(false);
console.log(`SONUÇ: ${rows.filter(Boolean).length}/${rows.length} beklenen; son Plan.visibility=${vis()}`);
