// Tarayıcının gönderdiği POST'un aynısı (Next-Action başlığı + encodeReply çok parçalı gövde); çerez yazdırılmaz.
import { execFileSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
const B = 'http://127.0.0.1:47993';
const ACTION = process.argv[3];
const COOKIE = readFileSync(process.argv[2], 'utf8').trim();
const psql = (sql) => execFileSync('docker', ['exec', 'devopfb93-pg', 'psql', '-U', 'fabrika', '-d', 'fabrika', '-v', 'ON_ERROR_STOP=1', '-tAc', sql]).toString().trim();
const MARK = 'devopfb93-internal-detail';
const TENANTS = psql(`select string_agg(id, '|') from "Tenant"`);
const LEAK = new RegExp(`devopfb93|internal-detail|SeatCandidate|ConsortiumSeat|invocation|prisma|P0001|P20\\d\\d|P10\\d\\d|23505|23503|constraint|tablo=|kolon=|not found|terminat|connection|Key \\(|fabrika|${TENANTS}`, 'i');

async function call(fields, locale = 'tr') {
  const fd = new FormData();
  // React encodeReply ile birebir (client.edge ile doğrulandı): alanlar `_1_` önekli ve ÖNCE; kök `0` EN SONDA.
  // Next gövdeyi akışla çözer — `0` önce gelirse FormData boş çözülür (ilk koşu bu yüzden geçersizdi).
  for (const [k, v] of Object.entries(fields)) fd.set(`_1_${k}`, v);
  fd.set('0', JSON.stringify([{ status: 'idle' }, '$K1']));
  const res = await fetch(`${B}/${locale}/main/opportunities/demo-opp-1/matching`, {
    method: 'POST',
    headers: { 'Next-Action': ACTION, Accept: 'text/x-component', Origin: B, Cookie: `authjs.session-token=${COOKIE}` },
    body: fd,
  });
  return { status: res.status, body: await res.text() };
}
const results = [];
function judge(label, r, expectCode) {
  const state = (r.body.match(/^1:(\{"status".*\})$/m) || [])[1] ?? '<yok>';
  const leak = LEAK.test(r.body);
  const ok = r.status === 200 && state === JSON.stringify({ status: 'error', code: expectCode }) && !leak;
  console.log(`[${label}] HTTP ${r.status} | durum=${state} | gövde ${r.body.length} bayt | sızıntı-eşleşme=${leak ? 'VAR' : 0}`);
  if (!ok) console.log('   gövde:', r.body.replace(/\s+/g, ' ').slice(0, 400));
  results.push([label, ok]);
}
const DEMO = { opportunityId: 'demo-opp-1', seatId: 'demo-seat-opp1-research' };
async function withTrigger(label, body, fn) {
  psql(`CREATE OR REPLACE FUNCTION devopfb93_boom() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN ${body}; RETURN NULL; END $$;
        CREATE TRIGGER devopfb93_boom BEFORE DELETE ON "SeatCandidate" FOR EACH STATEMENT EXECUTE FUNCTION devopfb93_boom();`);
  try { await fn(); } finally {
    psql('DROP TRIGGER IF EXISTS devopfb93_boom ON "SeatCandidate"; DROP FUNCTION IF EXISTS devopfb93_boom();');
    console.log(`   geri alındı (${label}): trigger=${psql(`select count(*) from pg_trigger where tgname='devopfb93_boom'`)}`);
  }
}

  console.log(`[sağlam] HTTP ${r.status} | durum=${st}`); results.push(['sağlam', r.status === 200 && /"status":"ok"/.test(st ?? '')]); }

// 1) P0001: iç ayrıntılı istisna (tablo/kolon/kiracı id'si içinde).
await withTrigger('P0001', `RAISE EXCEPTION '${MARK}: tablo=SeatCandidate kolon=seatId tenant=demo-tenant'`, async () => {
  for (const lc of ['tr', 'en']) judge(`${lc} P0001 iç metin`, await call(DEMO, lc), 'computeFailed');
});
// 2) Unique ihlali (23505 → Prisma P2002): kısıt adı + DETAIL'de kiracı id'si.
await withTrigger('23505', `RAISE EXCEPTION USING ERRCODE = '23505', MESSAGE = 'duplicate key value violates unique constraint "SeatCandidate_${MARK}_key"', DETAIL = 'Key ("tenantId", "seatId")=(demo-tenant, demo-seat-opp1-research) already exists.', CONSTRAINT = 'SeatCandidate_${MARK}_key', TABLE = 'SeatCandidate'`, async () => {
  judge('unique 23505', await call(DEMO), 'computeFailed');
});
// 3) FK ihlali (23503 → Prisma P2003).
await withTrigger('23503', `RAISE EXCEPTION USING ERRCODE = '23503', MESSAGE = 'insert or update on table "SeatCandidate" violates foreign key constraint "SeatCandidate_tenantId_fkey"', DETAIL = 'Key ("tenantId")=(demo-tenant) is not present in table "Tenant".', CONSTRAINT = 'SeatCandidate_tenantId_fkey', TABLE = 'SeatCandidate'`, async () => {
  judge('FK 23503', await call(DEMO), 'computeFailed');
});
// 4) Bağlantı kopması: sorgu ortasında backend kendini öldürür.
await withTrigger('terminate', `PERFORM pg_terminate_backend(pg_backend_pid())`, async () => {
  judge('bağlantı kopması', await call(DEMO), 'computeFailed');
});

for (const [n, ok] of results) console.log(`${ok ? 'GEÇTİ' : 'KALDI'}  ${n}`);
