// Logs every app-side pg Pool.connect; when FLAG holds a number k, lets k more
// connects through and fails the next one with pg-pool's own timeout text.
const fs = require('fs');
const FLAG = process.env.BEFB_PG_FLAG, LOG = process.env.BEFB_PG_LOG;
const pg = require(require.resolve('pg', { paths: [process.cwd()] }));
const orig = pg.Pool.prototype.connect;
pg.Pool.prototype.connect = function (cb) {
  let flag = null;
  try { flag = fs.readFileSync(FLAG, 'utf8').trim(); } catch {}
  const frames = (new Error().stack || '').split('\n').slice(2).map((l) => l.trim()).filter((l) => !l.includes('node_modules') && !l.includes('node:')).slice(0, 3).join(' <- ');
  let fail = false;
  if (flag !== null && flag !== '') {
    const k = parseInt(flag, 10);
    if (k <= 0) { fail = true; fs.unlinkSync(FLAG); } else fs.writeFileSync(FLAG, String(k - 1));
  }
  fs.appendFileSync(LOG, `${new Date().toISOString()} pid=${process.pid} max=${this.options && this.options.max} connect ${fail ? 'FAIL(injected)' : 'ok'} flag=${flag} ${frames}\n`);
  if (fail) {
    const e = new Error('timeout exceeded when trying to connect');
    if (typeof cb === 'function') { process.nextTick(() => cb(e)); return undefined; }
    return Promise.reject(e);
  }
  return orig.apply(this, arguments);
};
