// LOCAL PROOF ONLY — not committed. Real DB + real recordAlert → fake Sentry ingest.
import { prisma } from '@/lib/db';
import { recordAlert } from '@/lib/observability/system-alert';

async function main() {
  const source = 'se-fb-sentry-proof';
  const r = await recordAlert({
    level: 'error',
    source,
    message: `TASK-MULQMKP4YOM04 local proof ${Date.now()} — contact test.user@example.org`,
    meta: { userEmail: 'test.user@example.org', rows: 0, stage: 'k2-parse' },
  });
  console.log('recordAlert →', JSON.stringify(r));
  const row = await prisma.systemAlert.findUnique({ where: { id: r.id! }, select: { id: true, level: true, source: true, message: true, meta: true } });
  console.log('db row →', JSON.stringify(row));
  const warn = await recordAlert({ level: 'warn', source, message: 'warn must not be forwarded' });
  console.log('warn recordAlert →', JSON.stringify(warn));
  const del = await prisma.systemAlert.deleteMany({ where: { source } });
  console.log('cleanup deleted rows →', del.count);
  await prisma.$disconnect();
}
main();
