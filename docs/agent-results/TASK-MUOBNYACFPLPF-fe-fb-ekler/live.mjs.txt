// fe-fb TASK-MUOBNYACFPLPF — Y-1 canlı: prod next start + izole fefbny-pg; gerçek server-action POST.
import { execFileSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
import { encode } from 'next-auth/jwt';
const B = process.env.BASE; const LABEL = process.env.LABEL;
const man = JSON.parse(readFileSync('.next/server/app/[locale]/admin/cordis/page/server-reference-manifest.json', 'utf8'));
const idOf = (name) => Object.entries(man.node).find(([, v]) => v.exportedName === name)?.[0];
const psql = (sql) => execFileSync('docker', ['exec', 'fefbny-pg', 'psql', '-U', 'fabrika', '-d', 'fabrika', '-tAc', sql]).toString().trim();
const token = (id) => encode({ token: { id, sub: id }, secret: process.env.AUTH_SECRET, salt: 'authjs.session-token', maxAge: 3600 });
const admin = await token('demo-user-superadmin');
const LEAK = /fetch failed|ECONNREFUSED|CordisProject|\/srv\/secret|relation "/;
async function action(name, args) {
  const res = await fetch(`${B}/tr/admin/cordis`, { method: 'POST', body: JSON.stringify(args), headers: {
    'Next-Action': idOf(name), Accept: 'text/x-component', 'Content-Type': 'text/plain;charset=UTF-8', Origin: B, Cookie: `authjs.session-token=${admin}` } });
  const text = await res.text();
  return { status: res.status, line: (text.match(/^1:(.*)$/m) || [])[1] ?? text.slice(0, 200), leak: (text.match(LEAK) || [])[0] ?? null };
}
// 1) İçe aktarım: indirme proxy yüzünden düşer (sunucu NODE_USE_ENV_PROXY=1 HTTPS_PROXY=127.0.0.1:9 ile koşuyor)
const r1 = await action('handleRunCordisImportAction', ['HORIZON', 5]);
console.log(`[${LABEL}] handleRunCordisImportAction → HTTP ${r1.status} ${r1.line} | iç metin: ${r1.leak ?? 'YOK'}`);
console.log(`[${LABEL}]   DB'deki son koşu: ${psql(`select status||' | '||coalesce(error,'-') from "CordisImportRun" order by "startedAt" desc limit 1`)}`);
// 2) Durum okuma: son koşu satırına iç metin yaz, eylemi ve sayfayı oku
psql(`update "CordisImportRun" set error='relation "CordisProject" does not exist at /srv/secret/cordis.zip' where "runId"=(select "runId" from "CordisImportRun" order by "startedAt" desc limit 1)`);
const r2 = await action('handleGetK2Status', []);
console.log(`[${LABEL}] handleGetK2Status → HTTP ${r2.status} iç metin: ${r2.leak ?? 'YOK'}`);
const page = await (await fetch(`${B}/tr/admin/cordis`, { headers: { Cookie: `authjs.session-token=${admin}` } })).text();
console.log(`[${LABEL}] GET /tr/admin/cordis (SSR+RSC) → ${page.length} bayt, iç metin: ${(page.match(LEAK) || [])[0] ?? 'YOK'}, "Son Çalıştırma" var: ${page.includes('Son Çalıştırma')}`);
