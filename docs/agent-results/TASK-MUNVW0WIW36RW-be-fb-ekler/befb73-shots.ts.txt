// be-fb İNCELEME-73 — browser evidence, NOT committed. Real clicks on `next start` :3073.
import { chromium } from '@playwright/test';
import { encode } from 'next-auth/jwt';
import { prisma } from '../src/lib/db';
import { PLAN_E2E, seed } from './seed-plan-e2e';

const BASE = 'http://localhost:3073';
const OUT = process.env.SHOTS!;
const U = PLAN_E2E.ultra;
const COOKIE = 'authjs.session-token';
const KINDS: Record<string, string> = {
  nda: 'NDA', ndaSpace: 'NDA ', cyr: 'НDA', nbsp: 'NDA ', fullwidth: 'ＮＤＡ',
  lower: 'loi', contract: 'CONTRACT', esr: 'ESR_DEBRIEF', gone: 'AWARD_PACK',
};

async function main() {
  await seed();
  await prisma.membership.update({ where: { tenantId_userId: { tenantId: U.tenantId, userId: U.userId } }, data: { role: 'WRITER' } });
  const cookie = await encode({ token: { sub: U.userId, id: U.userId, name: 'i73 Writer', email: U.email, tenantId: U.tenantId, role: 'WRITER' }, secret: process.env.AUTH_SECRET!, salt: COOKIE, maxAge: 3600 });
  const browser = await chromium.launch({ executablePath: process.env.CHROME });
  const results: Record<string, unknown> = {};
  for (const [label, width, height] of [['390', 390, 844], ['1440', 1440, 900]] as const) {
    await prisma.followupDoc.deleteMany({ where: { tenantId: U.tenantId, opportunityId: U.opportunityId } });
    const ids: Record<string, string> = {};
    for (const [k, kind] of Object.entries(KINDS)) {
      ids[k] = (await prisma.followupDoc.create({ data: { tenantId: U.tenantId, opportunityId: U.opportunityId, kind, version: 1 }, select: { id: true } })).id;
    }
    const ctx = await browser.newContext({ viewport: { width, height } });
    await ctx.addCookies([{ name: COOKIE, value: cookie, url: BASE }]);
    await ctx.addInitScript(() => localStorage.setItem('fabrika_cookie_consent', JSON.stringify({ necessary: true, analytics: true, marketing: true })));
    const page = await ctx.newPage();
    await page.goto(`${BASE}/tr/${U.workspaceSlug}/opportunities/${U.opportunityId}/documents`, { waitUntil: 'networkidle' });
    const r: Record<string, unknown> = {};
    const btn: Record<string, number> = {};
    for (const k of Object.keys(KINDS)) btn[k] = await page.getByTestId(`sign-${ids[k]}`).count();
    r.signButtons = btn;
    r.rowCount = await page.locator('tbody tr').count();
    await page.screenshot({ path: `${OUT}/i73-${label}-table-top.png` });
    await page.getByTestId(`submit-approval-${ids.lower}`).scrollIntoViewIfNeeded();
    await page.screenshot({ path: `${OUT}/i73-${label}-table-bottom.png` });
    await page.screenshot({ path: `${OUT}/i73-${label}-table-full.png`, fullPage: true });
    // Y-1: refusal under the clicked row (row deleted after load → DOC_NOT_FOUND).
    await prisma.followupDoc.delete({ where: { id: ids.gone } });
    await page.getByTestId(`sign-${ids.gone}`).scrollIntoViewIfNeeded();
    await page.getByTestId(`sign-${ids.gone}`).click();
    const alert = page.getByTestId(`sign-error-${ids.gone}`);
    await alert.waitFor();
    r.refusalText = await alert.innerText();
    r.refusalInViewport = await alert.evaluate((el) => {
      const b = el.getBoundingClientRect();
      const y = b.top + b.height / 2;
      const a1 = document.elementFromPoint(b.left + 4, y); const a2 = document.elementFromPoint(b.right - 4, y);
      return b.top >= 0 && b.bottom <= innerHeight && b.left >= 0 && b.right <= innerWidth && !!a1 && !!a2 && el.contains(a1) && el.contains(a2);
    });
    await page.screenshot({ path: `${OUT}/i73-${label}-row-refusal.png` });
    // Counterfactual: direct kind signs with one click.
    await page.getByTestId(`sign-${ids.esr}`).scrollIntoViewIfNeeded();
    await page.getByTestId(`sign-${ids.esr}`).click();
    await page.waitForFunction((id) => !document.querySelector(`[data-testid="sign-${id}"]`), ids.esr);
    const rows = await prisma.followupDoc.findMany({ where: { tenantId: U.tenantId, opportunityId: U.opportunityId }, select: { id: true, kind: true, signedAt: true } });
    r.signedKinds = rows.filter((x) => x.signedAt).map((x) => x.kind);
    r.unsignedKinds = rows.filter((x) => !x.signedAt).map((x) => JSON.stringify(x.kind));
    await page.screenshot({ path: `${OUT}/i73-${label}-esr-signed.png` });
    results[label] = r;
    await ctx.close();
  }
  await browser.close();
  await prisma.followupDoc.deleteMany({ where: { tenantId: U.tenantId, opportunityId: U.opportunityId } });
  console.log(JSON.stringify(results, null, 2));
  await prisma.$disconnect();
}
main().catch((e) => { console.error(e); process.exit(1); });
