import { readFileSync } from 'node:fs';
const SP = process.argv[2]; const B = 'http://127.0.0.1:47979/api/cagrilar?lang=en';
const A = readFileSync(`${SP}/cookie.txt`, 'utf8'), Ch = readFileSync(`${SP}/cookie-charlie.txt`, 'utf8');
const get = (c) => fetch(B, { headers: { Cookie: c } });
const wait = (ms) => new Promise((r) => setTimeout(r, ms));
// Pin the shared window start: the first request after expiry resets it to "now".
// Exhaust alice first, so her next 200 can only come from a window reset.
await Promise.all(Array.from({ length: 31 }, () => get(A)));
let t0 = 0;
for (;;) { t0 = Date.now(); const r = await get(A); if (r.status === 200) { console.log(`alice isteği ${Date.now() - t0} ms sürdü`); break; } const ra = Number(r.headers.get('retry-after')); await wait(ra > 2 ? (ra - 2) * 1000 : 100); }
console.log('pencere başı sabitlendi (alice 200)');
await wait(58_800 - (Date.now() - t0));
const s = Date.now();
const a = await Promise.all(Array.from({ length: 30 }, () => get(Ch).then((r) => r.status)));
await wait(Math.max(0, 60_400 - (Date.now() - t0)));
const b = await Promise.all(Array.from({ length: 30 }, () => get(Ch).then((r) => r.status)));
const ok = [...a, ...b].filter((x) => x === 200).length;
console.log(`charlie: ${ok}/60 × 200, ${(Date.now() - s) / 1000}s içinde (eski pencerenin son ~0,7 sn'si: ${a.filter((x) => x === 200).length}/30, yeni pencere: ${b.filter((x) => x === 200).length}/30)`);
