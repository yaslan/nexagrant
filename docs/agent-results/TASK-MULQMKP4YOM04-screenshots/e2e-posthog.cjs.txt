// Real-browser proof: PostHog capture fires only with key + "accept all".
const { chromium } = require('/Users/macbook/alan/FONBRIDGE/node_modules/@playwright/test');
const SHOTS = '/Users/macbook/alan/FONBRIDGE/docs/agent-results/TASK-MULQMKP4YOM04-screenshots';

async function scenario(browser, base, label, consentButton) {
  const ctx = await browser.newContext({ locale: 'tr-TR' });
  const page = await ctx.newPage();
  const outbound = [];
  page.on('request', (r) => {
    const u = new URL(r.url());
    if (!['localhost'].includes(u.hostname) || /\/i\/v0\/e\/|envelope|posthog/.test(u.pathname + u.hostname))
      outbound.push({ method: r.method(), url: r.url(), body: r.postData() });
  });
  await page.goto(`${base}/tr/fon-hazirlik`, { waitUntil: 'domcontentloaded' });
  const btn = page.getByRole('button', { name: consentButton });
  await btn.waitFor({ timeout: 90000 });
  await btn.click();
  await page.locator('input[name="orgType"]').first().check();
  await page.getByRole('button', { name: 'Teste başla' }).click();
  await page.waitForTimeout(2500);
  await page.screenshot({ path: `${SHOTS}/${label}.png` });
  const storage = await page.evaluate(() => ({
    consent: localStorage.getItem('fabrika_cookie_consent'),
    anonId: localStorage.getItem('fabrika_analytics_anon_id'),
  }));
  await ctx.close();
  return { label, base, consentButton, outbound, storage };
}

(async () => {
  const browser = await chromium.launch({ executablePath: '/Users/macbook/Library/Caches/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-mac-arm64/chrome-headless-shell' });
  const results = [];
  results.push(await scenario(browser, 'http://localhost:3917', '01-key-set-essential-only', 'Gerekli Çerezler'));
  results.push(await scenario(browser, 'http://localhost:3917', '02-key-set-accept-all', 'Tümünü Kabul Et'));
  results.push(await scenario(browser, 'http://localhost:3000', '03-no-key-accept-all', 'Tümünü Kabul Et'));
  await browser.close();
  console.log(JSON.stringify(results, null, 2));
})().catch((e) => { console.error(e); process.exit(1); });
