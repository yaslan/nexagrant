async function makeTenant(tag: string) {
  const t = await admin.tenant.create({ data: { name: `${RUN} ${tag}`, slug: `${RUN}-${tag}`, country: 'TR' } });
  tenantIds.push(t.id);
  const ws = await admin.workspace.create({ data: { tenantId: t.id, name: 'w', slug: `${RUN}-ws-${tag}` } });
  const opp = await admin.opportunity.create({ data: { tenantId: t.id, workspaceId: ws.id, title: `${RUN} secret ${tag}` } });
  return { tenantId: t.id, oppId: opp.id };
}
async function makeKey(tenantId: string) {
  const raw = `fp_live_${RUN}_${crypto.randomBytes(8).toString('hex')}`;
  const key = await admin.apiKey.create({
    data: { tenantId, name: 'k', prefix: raw.slice(0, 12), keyHash: crypto.createHash('sha256').update(raw).digest('hex') },
  });
  return { raw, id: key.id };
}
const softDelete = (id: string, at: Date | null) => admin.tenant.update({ where: { id }, data: { deletedAt: at } });

type Hit = () => Promise<Response>;
function routes(raw: string, bearer = false): Record<string, Hit> {
  const h = bearer ? { Authorization: `Bearer ${raw}` } : { 'x-api-key': raw };
  const req = (p: string) => new Request(`http://localhost${p}`, { headers: h });
  return {
    opportunities: () => lib.opportunities.GET(req('/api/v1/opportunities')),
    calls: () => lib.calls.GET(req('/api/v1/calls')),
    partners: () => lib.partners.GET(req('/api/v1/partners')),
    'partners/[pic]': () =>
      lib.partner.GET(req('/api/v1/partners/999999999') as never, { params: Promise.resolve({ pic: '999999999' }) } as never),
  };
}

describe('A: API key, every /api/v1 route, both header forms', () => {
  test('A1 all four routes: live ≠ 401, deleted = 401 byte-identical to an unknown key, restored ≠ 401', async () => {
    const { tenantId } = await makeTenant('a1');
    const { raw } = await makeKey(tenantId);
    const unknown = await (await routes(`fp_live_nobody_${RUN}`).opportunities()).text();
    const log: string[] = [];
    for (const bearer of [false, true]) {
      for (const [name, hit] of Object.entries(routes(raw, bearer))) {
        await softDelete(tenantId, null);
        const live = await hit();
        await softDelete(tenantId, new Date());
        const dead = await hit();
        const deadBody = await dead.text();
        await softDelete(tenantId, null);
        const back = await hit();
        log.push(`${bearer ? 'Bearer' : 'x-api-key'} ${name}: live=${live.status} deleted=${dead.status} restored=${back.status}`);
        expect(live.status, name).not.toBe(401);
        expect(dead.status, name).toBe(401);
        expect(deadBody, name).toBe(unknown);
        expect(back.status, name).toBe(live.status);
      }
    }
    probe('[cs87 A1]\n' + log.join('\n'));
  });

  test('A2 hammering a deleted tenant’s key never reaches 429 and never moves the counter', async () => {
    const { tenantId } = await makeTenant('a2');
    const { raw, id } = await makeKey(tenantId);
    await softDelete(tenantId, new Date());
    const before = await admin.apiKey.findUniqueOrThrow({ where: { id } });
    const statuses = new Set<number>();
    for (let i = 0; i < 130; i++) statuses.add((await routes(raw).opportunities()).status);
    const after = await admin.apiKey.findUniqueOrThrow({ where: { id } });
    probe(`[cs87 A2] 130 calls → statuses ${[...statuses]}; count ${before.rateWindowCount}→${after.rateWindowCount}`);
    expect([...statuses]).toEqual([401]);
    expect(after.rateWindowCount).toBe(before.rateWindowCount);
    expect(after.lastUsedAt).toEqual(before.lastUsedAt);
  });

  test('A3 hard-purged tenant row (ApiKey orphan, no FK) → 401', async () => {
    const t = await admin.tenant.create({ data: { name: `${RUN} a3`, slug: `${RUN}-a3`, country: 'TR' } });
    const { raw, id } = await makeKey(t.id);
    await admin.tenant.delete({ where: { id: t.id } });
    const orphan = await admin.apiKey.findUnique({ where: { id } });
    const res = await routes(raw).opportunities();
    probe(`[cs87 A3] orphan key survives purge: ${orphan !== null}; status ${res.status}`);
    expect(res.status).toBe(401);
    await admin.apiKey.deleteMany({ where: { id } });
  });

  test('A4 (window) deletion lands after the key passed: the in-flight request still reads', async () => {
    const { tenantId } = await makeTenant('a4');
    const { raw } = await makeKey(tenantId);
    hooks.afterAuth = async (tid) => {
      await softDelete(tid, new Date());
    };
    const res = await routes(raw).opportunities();
    const body = await res.text();
    probe(`[cs87 A4] deleted between auth and read → status ${res.status}, secret in body: ${body.includes(`${RUN} secret a4`)}`);
    hooks.afterAuth = null;
    expect((await routes(raw).opportunities()).status).toBe(401); // the next request is closed
  });
});

describe('P: partner portal', () => {
  async function portal(tag: string) {
    const { tenantId, oppId } = await makeTenant(tag);
    const raw = lib.token.generateSecretToken();
    const part = await admin.applicationPart.create({
      data: { tenantId, opportunityId: oppId, part: 'PART_A', content: {}, magicTokenHash: lib.token.hashSecretToken(raw) },
    });
    const lines = async () => {
      const row = await admin.applicationPart.findUniqueOrThrow({ where: { id: part.id }, select: { content: true } });
      return ((row.content as { budget?: unknown[] } | null)?.budget ?? []).length;
    };
    return { tenantId, raw, lines };
  }

  test('P1 deleted tenant link is indistinguishable from an unknown token (write result, page, metadata)', async () => {
    const { tenantId, raw } = await portal('p1');
    const fake = lib.token.generateSecretToken();
    await softDelete(tenantId, new Date(Date.now() + 3_600_000)); // future deletedAt counts too
    const w = await lib.m11.addBudgetLineByToken(raw, 'X', 1);
    const wFake = await lib.m11.addBudgetLineByToken(fake, 'X', 1);
    const meta = await lib.portal.generateMetadata({ params: Promise.resolve({ token: raw, locale: 'en' }) });
    const metaFake = await lib.portal.generateMetadata({ params: Promise.resolve({ token: fake, locale: 'en' }) });
    const page = await lib.portal.default({ params: Promise.resolve({ token: raw, locale: 'en' }) }).then(
      () => 'rendered',
      (e) => (e === notFoundMarker ? 'notFound' : String(e))
    );
    probe(`[cs87 P1] write=${JSON.stringify(w)} fake=${JSON.stringify(wFake)} page=${page} metaTitle=${String(meta.title)}`);
    expect(w).toEqual(wFake);
    expect({ ...meta, alternates: undefined, openGraph: undefined }).toEqual({ ...metaFake, alternates: undefined, openGraph: undefined });
    expect(page).toBe('notFound');
  });

  test('P2 (window) deletion lands after discovery, before the write: does the write land?', async () => {
    const { tenantId, raw, lines } = await portal('p2');
    hooks.beforeWritableCheck = async (tid) => {
      await softDelete(tid, new Date());
    };
    const r = await lib.m11.addBudgetLineByToken(raw, 'RACE', 1);
    hooks.beforeWritableCheck = null;
    const n = await lines();
    probe(`[cs87 P2] deleted mid-action → result ${JSON.stringify(r)}, budget lines in deleted tenant: ${n}`);
    expect(await lib.m11.addBudgetLineByToken(raw, 'AFTER', 1)).toEqual({ success: false });
    await softDelete(tenantId, null);
  });
});
