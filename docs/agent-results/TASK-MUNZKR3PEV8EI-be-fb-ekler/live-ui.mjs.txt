// be-fb TASK-MUNZKR3PEV8EI: admin plan editor, real clicks against the prod build on befb-kr-pg (:55479).
// Faults are injected with triggers on the isolated DB only and removed afterwards. The cookie is never printed.
import { createRequire } from 'node:module';
import { execFileSync } from 'node:child_process';
import { readFileSync } from 'node:fs';

const SP = '/private/tmp/claude-501/-Users-macbook-alan/6036af88-425d-44cc-b342-e2b46a16e0fd/scratchpad';
const require = createRequire(`${SP}/wt/package.json`);
const { chromium } = require('@playwright/test');
const { encode } = require('next-auth/jwt');

const B = 'http://127.0.0.1:47988';
const psql = (sql) =>
  execFileSync('docker', ['exec', 'befb-kr-vt', 'psql', '-U', 'fabrika', '-d', 'fabrika', '-tAc', sql]).toString().trim();
const LEAK = /befbkr|tablo=|kolon=|P20\d\d|23505|prisma|invocation|constraint|Unique|\{detail\}/i;

const salt = 'authjs.session-token';
const value = await encode({
  secret: readFileSync(`${SP}/authsecret`, 'utf8').trim(),
  salt,
  maxAge: 3600,
  token: { sub: 'demo-user-superadmin', id: 'demo-user-superadmin', email: 'admin@yasinaslan.com.tr' },
});

const browser = await chromium.launch({ executablePath: '/Users/macbook/Library/Caches/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-mac-arm64/chrome-headless-shell' });
const ctx = await browser.newContext();
await ctx.addCookies([{ name: salt, value, url: B }]);
await ctx.addInitScript(() =>
  localStorage.setItem('fabrika_cookie_consent', JSON.stringify({ necessary: true, analytics: true, marketing: true }))
);
const page = await ctx.newPage();
let fails = 0;
const check = (label, ok, extra = '') => {
  if (!ok) fails++;
  console.log(`${ok ? 'OK  ' : 'FAIL'} ${label}${extra ? ` — ${extra}` : ''}`);
};

async function scenario(locale, name, fault, act, expectText, shot) {
  psql(fault.install);
  try {
    await page.goto(`${B}/${locale}/admin/billing/plans`);
    await page.getByTestId('plan-row-ultra').click();
    await act();
    const notice = page.getByTestId('plan-notice');
    await notice.waitFor({ timeout: 15000 });
    const text = (await notice.innerText()).trim();
    console.log(`-- [${locale}] ${name}: plan-notice = "${text}"`);
    check(`[${locale}] ${name}: expected sentence`, text === expectText);
    check(`[${locale}] ${name}: no internal text on screen`, !LEAK.test(await page.locator('body').innerText()));
    await page.screenshot({ path: `${SP}/shots/${shot}.png`, fullPage: true });
  } finally {
    psql(fault.remove);
  }
}

const plainFault = {
  install: `CREATE OR REPLACE FUNCTION befbkr_boom() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN RAISE EXCEPTION 'befbkr-internal-detail: tablo=Plan kolon=visibility'; END $$;
            CREATE TRIGGER befbkr_boom BEFORE UPDATE ON "Plan" FOR EACH ROW EXECUTE FUNCTION befbkr_boom();`,
  remove: `DROP TRIGGER befbkr_boom ON "Plan"; DROP FUNCTION befbkr_boom();`,
};
const uniqueFault = {
  install: `CREATE OR REPLACE FUNCTION befbkr_dup() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN RAISE EXCEPTION 'befbkr-dup: tablo=PlanVersion' USING ERRCODE = 'unique_violation'; END $$;
            CREATE TRIGGER befbkr_dup BEFORE INSERT ON "PlanVersion" FOR EACH ROW EXECUTE FUNCTION befbkr_dup();`,
  remove: `DROP TRIGGER befbkr_dup ON "PlanVersion"; DROP FUNCTION befbkr_dup();`,
};
const tr = JSON.parse(readFileSync(`${SP}/wt/messages/tr.json`, 'utf8')).adminPlans.errors;
const en = JSON.parse(readFileSync(`${SP}/wt/messages/en.json`, 'utf8')).adminPlans.errors;

const visibility = () => page.getByTestId('visibility-select').selectOption('private');
const createDraft = () => page.getByTestId('create-draft').click();

await scenario('tr', 'görünürlük (UPDATE hata tetikleyicisi)', plainFault, visibility, tr.UNKNOWN, 'tr-visibility-unknown');
await scenario('en', 'visibility (UPDATE fault trigger)', plainFault, visibility, en.UNKNOWN, 'en-visibility-unknown');
await scenario('tr', 'taslak oluştur (INSERT unique_violation)', uniqueFault, createDraft, tr.UNKNOWN, 'tr-createdraft-unique');
await scenario('en', 'create draft (INSERT unique_violation)', uniqueFault, createDraft, en.UNKNOWN, 'en-createdraft-unique');

// Page load: the read fails (table renamed) — the screen says so, without the database's text.
for (const locale of ['tr', 'en']) {
  psql(`ALTER TABLE "Plan" RENAME TO "Plan_befbkr"`);
  try {
    await page.goto(`${B}/${locale}/admin/billing/plans`);
    const err = page.getByTestId('plans-load-error');
    await err.waitFor({ timeout: 15000 });
    const body = await page.locator('main').innerText();
    console.log(`-- [${locale}] load failure: plans-load-error = "${(await err.innerText()).trim()}"`);
    check(`[${locale}] load failure: no internal text on screen`, !LEAK.test(body) && !/Plan_befbkr|42P01|relation/i.test(body));
    await page.screenshot({ path: `${SP}/shots/${locale}-load-error.png`, fullPage: true });
  } finally {
    psql(`ALTER TABLE "Plan_befbkr" RENAME TO "Plan"`);
  }
}

// Nothing changed: both faults rolled the write back.
check('Plan ultra still public', psql(`select visibility from "Plan" where id='ultra'`) === 'public');
check('no draft left on ultra', psql(`select count(*) from "PlanVersion" where "planId"='ultra' and status='draft'`) === '0');
check(
  'triggers removed',
  psql(`select count(*) from pg_trigger where tgname in ('befbkr_boom','befbkr_dup')`) === '0'
);
await browser.close();
console.log(fails === 0 ? 'ALL OK' : `${fails} FAILED`);
process.exit(fails === 0 ? 0 : 1);
