// Leaf-ish elements carrying the token: is the text clipped/hidden instead of wrapped?
async function hiddenCheck(p: Page, tok: string) {
  return p.evaluate((tok) => {
    const main = document.querySelector('main')!; const mr = main.getBoundingClientRect(); const res: string[] = [];
    const probe = tok.slice(0, 20);
    for (const el of document.querySelectorAll<HTMLElement>('main *, [class*="fixed"] *')) {
      if (!(el.textContent ?? '').includes(probe)) continue;
      if ([...el.children].some((c) => (c.textContent ?? '').includes(probe))) continue;
      const cs = getComputedStyle(el); const r = el.getBoundingClientRect();
      const tag = `${el.tagName.toLowerCase()}`;
      if (cs.textOverflow === 'ellipsis') res.push(`${tag} ellipsis`);
      if (el.scrollWidth > el.clientWidth + 1 && el.clientWidth > 0) res.push(`${tag} sw>cw ${el.scrollWidth}/${el.clientWidth}`);
      if (r.right > Math.min(mr.right, innerWidth) + 1) res.push(`${tag} right=${Math.round(r.right)}>${Math.round(Math.min(mr.right, innerWidth))}`);
      for (let a = el.parentElement; a && a !== document.body; a = a.parentElement) {
        const ox = getComputedStyle(a).overflowX;
        if ((ox === 'hidden' || ox === 'clip') && a.scrollWidth > a.clientWidth + 1) res.push(`${tag} clipped-by ${a.tagName.toLowerCase()}.${String(a.className).slice(0, 40)}`);
      }
    }
    return res.length ? res.slice(0, 3).join(' ; ') : 'görünür';
  }, tok);
}
async function culprit(p: Page) {
  return p.evaluate(() => {
    const main = document.querySelector('main')!; const mr = main.getBoundingClientRect(); const out: string[] = [];
    for (const el of main.querySelectorAll<HTMLElement>('*')) {
      const over = el.getBoundingClientRect().right - mr.right; if (over <= 0.5) continue;
      let inner = false; for (let e = el.parentElement; e && e !== main; e = e.parentElement) if (getComputedStyle(e).overflowX !== 'visible') { inner = true; break; }
      if (inner) continue;
      const par = el.parentElement!; if (par !== main && par.getBoundingClientRect().right - mr.right >= over - 0.5) continue;
      out.push(`${el.tagName.toLowerCase()}.${String(el.className).slice(0, 50)} +${Math.round(over)}`);
    }
    return out.slice(0, 2).join(' | ');
  });
}

    const over = r.sw - r.cw; let hid = await hiddenCheck(p, tok);
    if (name === 'firsat-arsiv-onay') {
      // The band itself: at 390 İNCELEME-109 saw <main> equal while the text ran out of the amber box.
      const band = await p.getByTestId('pipeline-archive-confirm').evaluate((el) => {
        const b = el.getBoundingClientRect(); const t = el.querySelector('#archive-confirm-title')!;
        let right = 0; for (const n of t.querySelectorAll('*')) right = Math.max(right, n.getBoundingClientRect().right);
        const rg = document.createRange(); rg.selectNodeContents(t); for (const q of rg.getClientRects()) right = Math.max(right, q.right);
        return { sw: el.scrollWidth - el.clientWidth, spill: Math.round(right - (b.right - 16)) };
      });
      if (band.sw > 0 || band.spill > 1) hid = (hid === 'görünür' ? '' : hid + ' ; ') + `bant sw-cw=${band.sw} metin-bant-dışı=+${band.spill}`;
      else hid += ` bant=OK(sw-cw=0,iç)`;
      if (hid.endsWith('bant=OK(sw-cw=0,iç)') && hid.startsWith('görünür')) hid = 'görünür bant=OK';
    }
    const bad = over || !hid.startsWith('görünür') || r.dsw > r.iw;
    if (bad) fails++; perScreen[name] = (perScreen[name] ?? 0) + (bad ? 1 : 0);

    if (SHOT_W.includes(w)) await p.screenshot({ path: `${out}/kareler/${label}-${v}-${name}-${w}.png` });
  }
}
