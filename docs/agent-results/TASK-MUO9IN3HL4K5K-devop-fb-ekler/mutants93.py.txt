def vit(paths):
    r = subprocess.run(['npx', 'vitest', 'run', *paths], cwd=W, capture_output=True, text=True)
    out = r.stdout + r.stderr
    fails = sorted({l.strip()[:160] for l in out.splitlines() if l.strip().startswith(('×', 'FAIL', '✗'))})
    return r.returncode, fails
