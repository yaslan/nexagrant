import { readdirSync, readFileSync, statSync } from 'node:fs';
import { join } from 'node:path';
import { actionReturnLeaks, hasUseServerDirective } from '../tests/api/action-leak-scan.ts';
const walk = (d: string): string[] => readdirSync(d).flatMap((n) => { const f = join(d, n); return statSync(f).isDirectory() ? walk(f) : /\.tsx?$/.test(n) && !/\.test\.tsx?$/.test(n) ? [f] : []; });
for (let i = 0; i < 3; i++) {
  const t = performance.now(), c = process.cpuUsage();
  const files = walk('src').filter((f) => hasUseServerDirective(readFileSync(f, 'utf8'), f));
  let leaks = 0; for (const f of files) leaks += actionReturnLeaks(readFileSync(f, 'utf8'), f).length;
  const u = process.cpuUsage(c);
  console.log(`koşu ${i + 1}: eylem dosyası=${files.length} sızıntı=${leaks} duvar=${(performance.now() - t).toFixed(0)} ms cpu=${((u.user + u.system) / 1000).toFixed(0)} ms`);
}
