// İNCELEME-86 (be-fb) scratch probe — NOT committed. Isolated DB only.
import { afterAll, beforeAll, beforeEach, describe, expect, test, vi } from 'vitest';
import type { PrismaClient } from '@prisma/client';

const RUN = `rv86-${Date.now().toString(36)}`;
const COOKIE = 'authjs.session-token';
const jar = vi.hoisted(() => new Map<string, string>());
vi.mock('next/headers', () => ({
  headers: async () =>
    new Headers({ host: 'localhost:3000', 'x-forwarded-proto': 'http', cookie: [...jar].map(([k, v]) => `${k}=${v}`).join('; ') }),
  cookies: async () => ({
    get: (name: string) => (jar.has(name) ? { name, value: jar.get(name)! } : undefined),
    set: (name: string, value: string) => void (value ? jar.set(name, value) : jar.delete(name)),
    delete: (name: string) => void jar.delete(name),
  }),
}));
vi.mock('next/cache', () => ({ revalidatePath: () => {}, revalidateTag: () => {} }));
vi.mock('next-intl/server', () => ({ getTranslations: async () => (key: string) => key }));

let admin: PrismaClient;
let auth: typeof import('../src/lib/auth');
let encode: typeof import('next-auth/jwt').encode;
let token: typeof import('../src/lib/secret-token');
let acceptActions: typeof import('../src/app/[locale]/accept-invite/continue/actions');
let acceptCookie: typeof import('../src/lib/invitations/accept-cookie');
const tenants: string[] = [];
const users: string[] = [];
const ids = { a: '', d: '', u: '', email: '' };

beforeAll(async () => {
  if (!process.env.DATABASE_URL?.includes(':5486/')) throw new Error('isolated DB only');
  admin = (await import('../src/lib/db')).prisma;
  auth = await import('../src/lib/auth');
  encode = (await import('next-auth/jwt')).encode;
  token = await import('../src/lib/secret-token');
  acceptActions = await import('../src/app/[locale]/accept-invite/continue/actions');
  acceptCookie = await import('../src/lib/invitations/accept-cookie');
});

async function mkTenant(label: string, deleted = false) {
  const t = await admin.tenant.create({
    data: { name: `${RUN} ${label}`, slug: `${RUN}-${label}-${Math.random().toString(36).slice(2, 7)}`, country: 'TR', deletedAt: deleted ? new Date() : null },
  });
  tenants.push(t.id);
  return t.id;
}
async function signIn(userId: string, email: string) {
  const claims = await auth.authConfig.callbacks!.jwt!({ token: { sub: userId, email }, user: { id: userId, email }, trigger: 'signIn' } as never);
  jar.clear();
  jar.set(COOKIE, await encode({ token: claims as never, secret: process.env.AUTH_SECRET!, salt: COOKIE }));
}
async function req() {
  const s = (await auth.auth()) as { tenantId: string | null; role: string | null; isPlatformAdmin?: boolean } | null;
  return { t: s?.tenantId === ids.a ? 'A' : s?.tenantId === ids.d ? 'D' : s?.tenantId ?? null, role: s?.role ?? null, admin: s?.isPlatformAdmin ?? null };
}
const soft = (id: string, at: Date | null) => admin.tenant.update({ where: { id }, data: { deletedAt: at } });

beforeEach(async () => {
  ids.a = await mkTenant('a');
  ids.d = await mkTenant('d');
  ids.email = `${RUN}-${Math.random().toString(36).slice(2, 7)}@example.invalid`;
  const u = await admin.user.create({ data: { email: ids.email, name: 'U' } });
  ids.u = u.id;
  users.push(u.id);
  await admin.membership.create({ data: { tenantId: ids.a, userId: u.id, role: 'TENANT_OWNER', createdAt: new Date(Date.now() - 120_000) } });
  await admin.membership.create({ data: { tenantId: ids.d, userId: u.id, role: 'TENANT_OWNER', createdAt: new Date(Date.now() - 60_000) } });
  await signIn(u.id, ids.email);
});

afterAll(async () => {
  for (const id of tenants) await admin.tenant.delete({ where: { id } }).catch(() => {});
  for (const id of users) await admin.user.delete({ where: { id } }).catch(() => {});
});

describe('jwt probes', () => {
  test('J1 platform admin bound to a deleted tenant keeps admin, loses tenant', async () => {
    await admin.user.update({ where: { id: ids.u }, data: { isPlatformAdmin: true } });
    const before = await req();
    await soft(ids.a, new Date());
    const after = await req();
    console.log('J1', JSON.stringify({ before, after }));
    expect(after).toMatchObject({ t: null, role: null, admin: true });
  });

  test('J2 every membership deleted: sign-in tenantless; restore → back', async () => {
    await soft(ids.a, new Date());
    await soft(ids.d, new Date());
    await signIn(ids.u, ids.email);
    const s = await req();
    await soft(ids.a, null);
    const plain = await req(); // no binding from sign-in → oldest live on next call
    console.log('J2', JSON.stringify({ s, plain }));
    expect(s.t).toBeNull();
  });

  test('J3 bound A deleted; explicit update to live D moves there; update back to deleted A refused', async () => {
    await soft(ids.a, new Date());
    const stranded = await req();
    await auth.unstable_update({ tenantId: ids.d } as never);
    const moved = await req();
    await auth.unstable_update({ tenantId: ids.a } as never);
    const refused = await req();
    console.log('J3', JSON.stringify({ stranded, moved, refused }));
    expect(stranded.t).toBeNull();
    expect(moved.t).toBe('D');
    expect(refused.t).toBe('D');
  });

  test('J4 invitation to a deleted tenant: accept refused, session unchanged', async () => {
    const c = await mkTenant('c', true);
    const raw = token.generateSecretToken();
    await admin.invitation.create({ data: { email: ids.email, tenantId: c, role: 'WRITER', tokenHash: token.hashSecretToken(raw), expires: new Date(Date.now() + 86_400_000) } });
    jar.set(acceptCookie.INVITATION_ACCEPT_COOKIE, acceptCookie.encodeParkedInvitation({ token: raw, locale: 'tr' }));
    let out: string;
    try { out = `ok:${JSON.stringify(await acceptActions.acceptInvitationAction('tr'))}`; } catch (e) { out = `throw:${(e as { digest?: string; message?: string }).digest ?? (e as Error).message}`; }
    const after = await req();
    const joined = await admin.membership.count({ where: { tenantId: c, userId: ids.u } });
    console.log('J4', JSON.stringify({ out: out.slice(0, 120), after, joined }));
    expect(after.t).toBe('A');
    expect(joined).toBe(0);
  });
});

describe('sessionless tenant paths (not covered by the fix)', () => {
  test('P1 API key of a soft-deleted tenant', async () => {
    const raw = `fp_live_${RUN}_${Math.random().toString(36).slice(2)}`;
    const crypto = await import('crypto');
    await admin.apiKey.create({ data: { tenantId: ids.d, name: 'k', keyHash: crypto.createHash('sha256').update(raw).digest('hex'), prefix: raw.slice(0, 12) } });
    const ws = await admin.workspace.create({ data: { tenantId: ids.d, name: 'w', slug: 'default' } });
    await admin.opportunity.create({ data: { tenantId: ids.d, workspaceId: ws.id, title: `${RUN} secret opp` } });
    const { GET } = await import('../src/app/api/v1/opportunities/route');
    const live = await GET(new Request('http://localhost/api/v1/opportunities', { headers: { 'x-api-key': raw } }));
    await soft(ids.d, new Date());
    const del = await GET(new Request('http://localhost/api/v1/opportunities', { headers: { 'x-api-key': raw } }));
    const body = (await del.json()) as { meta?: { total?: number }; data?: Array<{ title: string }> };
    console.log('P1', JSON.stringify({ live: live.status, deleted: del.status, total: body.meta?.total, title: body.data?.[0]?.title }));
    expect(live.status).toBe(200);
  });

  test('P2 partner portal magic token of a soft-deleted tenant', async () => {
    const ws = await admin.workspace.create({ data: { tenantId: ids.d, name: 'w', slug: 'default' } });
    const opp = await admin.opportunity.create({ data: { tenantId: ids.d, workspaceId: ws.id, title: `${RUN} opp` } });
    const raw = token.generateSecretToken();
    const part = await admin.applicationPart.create({ data: { tenantId: ids.d, opportunityId: opp.id, part: 'PART_A', content: {}, magicTokenHash: token.hashSecretToken(raw) } });
    await soft(ids.d, new Date());
    const { addBudgetLineByToken } = await import('../src/modules/m11/server/actions');
    let out: string;
    try { out = `ok:${JSON.stringify(await addBudgetLineByToken(raw, 'ACME', 2))}`; } catch (e) { out = `throw:${(e as Error).message}`; }
    const after = await admin.applicationPart.findUniqueOrThrow({ where: { id: part.id }, select: { content: true } });
    console.log('P2', JSON.stringify({ out, budget: (after.content as { budget?: unknown[] })?.budget?.length ?? 0 }));
    expect(true).toBe(true);
  });
});
