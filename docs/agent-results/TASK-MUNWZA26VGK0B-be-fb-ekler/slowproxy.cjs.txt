// TCP proxy to befbwz-pg: every Nth new connection's first server bytes are held DELAY ms
// (a slow handshake, as on a loaded host). Usage: node slowproxy.cjs <listen> <every> <delayMs>
const net = require('net');
const [listen, every, delay] = process.argv.slice(2).map(Number);
let n = 0;
net.createServer((c) => {
  const slow = ++n % every === 0; console.log('conn', n, slow ? 'SLOW' : '');
  const up = net.connect(55491, '127.0.0.1');
  let held = slow;
  if (slow) setTimeout(() => { held = false; if (buf.length) { c.write(Buffer.concat(buf)); buf = []; } }, delay);
  let buf = [];
  up.on('data', (d) => (held ? buf.push(d) : c.write(d)));
  c.on('data', (d) => up.write(d));
  const end = () => { c.destroy(); up.destroy(); };
  c.on('error', end); up.on('error', end); c.on('close', end); up.on('close', end);
}).listen(listen, '127.0.0.1', () => console.log('proxy', listen, 'slow every', every, 'delay', delay));
