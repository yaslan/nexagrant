def run(files=T):
    r=subprocess.run(['pnpm','exec','vitest','run']+files.split(),cwd=WT,capture_output=True,text=True)
    out=r.stdout+r.stderr
    line=[l for l in out.splitlines() if l.strip().startswith('Tests ')]
    return (line[-1].strip() if line else out[-600:])
print('BASE', run())
