import { test, vi } from 'vitest';
vi.mock('next/headers', () => ({ headers: async () => new Headers(), cookies: async () => ({ get: () => undefined, set: () => {}, delete: () => {} }) }));
test('o2 burst through slow proxy', async () => {
  const { authConfig } = await import('../src/lib/auth');
  const jwt = authConfig.callbacks.jwt as (a: unknown) => Promise<Record<string, unknown>>;
  const warn = vi.spyOn(console, 'warn');
  for (let round = 1; round <= 1; round++) {
    await new Promise((r) => setTimeout(r, 6000)); // idleTimeoutMillis 5000: the pool is cold again
    // Staggered by a few ms, as HTTP requests arrive: Prisma batches findUnique calls made in the same tick.
    const out = await Promise.all(
      Array.from({ length: 20 }, (_, i) =>
        new Promise((r) => setTimeout(r, i * 3)).then(() => jwt({ token: { id: 'wz-u', tenantId: 'wz-b', boundTenantId: 'wz-b' } }))
      )
    );
    const counts: Record<string, number> = {};
    for (const t of out) counts[String(t.tenantId)] = (counts[String(t.tenantId)] ?? 0) + 1;
    console.log(`O2 round ${round}:`, JSON.stringify(counts));
  }
  const msgs = warn.mock.calls.map((c) => String(c[0])).filter((m) => m.includes('[Auth.jwt]'));
  console.log('O2 warnings:', msgs.length, JSON.stringify([...new Set(msgs)]));
}, 120000);
