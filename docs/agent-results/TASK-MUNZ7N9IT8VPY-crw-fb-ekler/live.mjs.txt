// Hata enjeksiyonu YALNIZ crwfb7n-pg'de; her senaryodan sonra geri alınır. Çerez yazdırılmaz.
import { chromium } from '@playwright/test';
import { execFileSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
const B = 'http://127.0.0.1:47978';
const COOKIE = readFileSync(process.argv[2], 'utf8').trim();
const psql = (sql) => execFileSync('docker', ['exec', 'crwfb7n-pg', 'psql', '-U', 'fabrika', '-d', 'fabrika', '-tAc', sql]).toString().trim();
const MARK = 'crwfb7n-internal-detail';
const LEAK = /crwfb7n|internal-detail|SeatCandidate|invocation|prisma|P2010|P0001|tablo=|Seat .* not found|Opportunity .* not found|fabrika/i;
const cat = { tr: JSON.parse(readFileSync('messages/tr.json', 'utf8')).Consortium, en: JSON.parse(readFileSync('messages/en.json', 'utf8')).Consortium };

const browser = await chromium.launch(process.env.PW_EXE ? { executablePath: process.env.PW_EXE } : {});
async function run(locale, seatId, label) {
  const ctx = await browser.newContext();
  await ctx.addCookies([{ name: 'authjs.session-token', value: COOKIE, url: B }]);
  const page = await ctx.newPage();
  // Sunucu eyleminin yanıt gövdesi (RSC akışı) istek BİTİNCE okunur; boş gövde kanıt sayılmaz.
  const pending = [];
  page.on('requestfinished', (req) => {
    if (req.method() === 'POST') pending.push(req.response().then((r) => r.body()).then((b) => b.toString('utf8')).catch((e) => `<okunamadı: ${e}>`));
  });
  await page.goto(`${B}/${locale}/main/opportunities/demo-opp-1/matching`);
  // Çerez onay penceresi (alertdialog) tıklamaları keser: yalnız zorunlu çerezlerle kapat.
  const consent = page.getByRole('alertdialog');
  // Hidrasyondan sonra açılır: hemen bakmak yarışa düşer, 8 sn'ye kadar bekle.
  if (await consent.waitFor({ timeout: 8000 }).then(() => true, () => false)) {
    await consent.getByRole('button').last().click();
    await consent.waitFor({ state: 'hidden' });
  }
  return { ctx, page, async click() {
    const form = page.locator(`form:has(input[name="seatId"][value="${seatId}"])`);
    await form.getByRole('button', { name: cat[locale].findCandidates }).click();
    const alert = form.getByRole('alert');
    await alert.waitFor({ timeout: 15000 });
    const text = (await alert.textContent()).trim();
    await page.waitForTimeout(300);
    const bodies = await Promise.all(pending);
    const empty = bodies.filter((b) => !/"status":"error"/.test(b)).length;
    const leaks = bodies.filter((b) => LEAK.test(b)).length;
    console.log(`[${label}] alert="${text}" | POST gövdesi=${bodies.length} (durum içermeyen=${empty}) | sızıntı-eşleşen gövde=${leaks}`);
    for (const b of bodies) console.log(`   gövde: ${b.replace(/\s+/g, ' ').slice(0, 220)}`);
    await ctx.close();
    return { text, leaks: leaks + empty + (bodies.length === 0 ? 1 : 0) };
  } };
}

const results = [];
// 1-2: canlı Prisma hatası — SeatCandidate üzerinde her DELETE'te iç ayrıntılı istisna.
psql(`CREATE OR REPLACE FUNCTION crwfb7n_boom() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN RAISE EXCEPTION '${MARK}: tablo=SeatCandidate kolon=seatId'; END $$;
      CREATE TRIGGER crwfb7n_boom BEFORE DELETE ON "SeatCandidate" FOR EACH STATEMENT EXECUTE FUNCTION crwfb7n_boom();`);
console.log('== enjeksiyon: statement trigger SeatCandidate BEFORE DELETE ->', psql(`select count(*) from pg_trigger where tgname='crwfb7n_boom'`));
try {
  for (const lc of ['tr', 'en']) {
    const r = await (await run(lc, 'demo-seat-opp1-research', `${lc} prisma-hatası`)).click();
    results.push([`${lc} prisma`, r.text === cat[lc].computeFailed && r.leaks === 0]);
  }
} finally {
  psql('DROP TRIGGER IF EXISTS crwfb7n_boom ON "SeatCandidate"; DROP FUNCTION IF EXISTS crwfb7n_boom();');
  console.log('== geri alındı: trigger sayısı =', psql(`select count(*) from pg_trigger where tgname='crwfb7n_boom'`));
}
// 3-4: alan hatası — sayfa açıkken koltuk silinir (başka sekmede silinmiş gibi).
for (const lc of ['tr', 'en']) {
  const id = `crwfb7n-tmp-seat-${lc}`;
  psql(`INSERT INTO "ConsortiumSeat"(id,"tenantId","opportunityId","roleKey",required,status) VALUES ('${id}','demo-tenant','demo-opp-1','industry',true,'OPEN') ON CONFLICT DO NOTHING`);
  const s = await run(lc, id, `${lc} koltuk-silinmiş`);
  psql(`DELETE FROM "ConsortiumSeat" WHERE id='${id}'`);
  const r = await s.click();
  results.push([`${lc} seatNotFound`, r.text === cat[lc].seatNotFound && r.leaks === 0]);
}
// 5: sağlam yol hâlâ çalışıyor (hata yok, alert yok).
{
  const s = await run('tr', 'demo-seat-opp1-research', 'tr sağlam');
  await s.page.locator('form:has(input[name="seatId"][value="demo-seat-opp1-research"])').getByRole('button', { name: cat.tr.findCandidates }).click();
  await s.page.waitForResponse((r) => r.request().method() === 'POST');
  await s.page.waitForTimeout(800);
  const alerts = await s.page.locator('form:has(input[name="seatId"][value="demo-seat-opp1-research"]) [role=alert]').count();
  console.log(`[tr sağlam] alert sayısı=${alerts}`);
  results.push(['tr sağlam', alerts === 0]);
  await s.ctx.close();
}
await browser.close();
for (const [n, ok] of results) console.log(`${ok ? 'GEÇTİ' : 'KALDI'}  ${n}`);
process.exit(results.every(([, ok]) => ok) ? 0 : 1);
