import re,os,glob
acts=[]
for dp,_,fs in os.walk('src'):
    for f in fs:
        if re.search(r'\.tsx?$',f) and not f.endswith('.test.ts'):
            p=os.path.join(dp,f); s=open(p).read()
            if re.match(r"\s*(//[^\n]*\n\s*|/\*[\s\S]*?\*/\s*)*['\"]use server['\"]",s): acts.append(p)
print('actions',len(acts))
for p in sorted(acts):
    s=open(p).read()
    imported=set()
    for m in re.finditer(r"import\s+(?!type)\{([^}]*)\}\s+from\s+['\"](@/|\.)[^'\"]*['\"]",s):
        for n in re.findall(r"(?:^|,)\s*(?:type\s+)?([A-Za-z_$][\w$]*)(?:\s+as\s+([\w$]+))?",m.group(1)):
            imported.add(n[1] or n[0])
    lines=s.split('\n')
    for i,l in enumerate(lines,1):
        m=re.search(r"\breturn\s+(await\s+)?([A-Za-z_$][\w$]*)\(",l)
        hit=None
        if m and m.group(2) in imported and m.group(2) not in ('requirePlatformAdmin','requireTenantAction','t'): hit='direct-return '+m.group(2)
        elif re.search(r"\b(return|error|message|reason)\b[^;]*\b\w+\.(error|reason|message|details?)\b",l) and 'console' not in l and 'logger' not in l: hit='field-pass'
        elif re.search(r"\.\.\.\s*\w+(Res|Result|result|res)\b",l): hit='spread'
        if hit: print(f"{p.replace('src/app/[locale]/','')}:{i}: [{hit}] {l.strip()[:150]}")
