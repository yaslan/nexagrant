// yüklenir; cordis.europa.eu isteğini $CORDIS_MODE_FILE içindeki moda göre yönlendirir.
import { readFileSync } from 'node:fs';
const real = globalThis.fetch;
globalThis.fetch = async function hooked(input, init) {
  const url = typeof input === 'string' ? input : input?.url ?? String(input);
  if (!url.startsWith('https://cordis.europa.eu/')) return real(input, init);
  const mode = readFileSync(process.env.CORDIS_MODE_FILE, 'utf8').trim();
  if (mode === 'refused') return real('http://127.0.0.1:9/cordis.zip');                  // gerçek ECONNREFUSED → "fetch failed"
  if (mode === 'timeout') return real('http://10.255.255.1/cordis.zip', { signal: AbortSignal.timeout(1500) }); // gerçek TimeoutError
  if (mode === 'http503') return new Response('upstream down', { status: 503, statusText: 'Service Unavailable' });
  if (mode.startsWith('file:')) return new Response(readFileSync(mode.slice(5)));
  throw new Error('fetch-hook: bilinmeyen mod ' + mode);
};
