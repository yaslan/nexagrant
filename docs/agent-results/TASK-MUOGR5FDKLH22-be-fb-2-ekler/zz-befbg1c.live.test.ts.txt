/** Is the confirm lock of this final key held by someone? (admin session, try-lock in its own tx) */
async function lockHeld(finalKey: string) {
  return admin.$transaction(async (tx) => {
    const [r] = await tx.$queryRaw<{ got: boolean }[]>`SELECT pg_try_advisory_xact_lock(${lockKeyOf(finalKey)}) AS got`;
    return !r!.got;
  });
}

}

/** What the caller of the action is left holding, in one word. */
function classify(e: unknown): string {
  const err = e as { message?: unknown; digest?: unknown; code?: unknown };
  const msg = String(e instanceof Error ? e.message : e).split('\n').slice(-1)[0]!.slice(0, 60);
  const tag = typeof err?.digest === 'string' ? `digest=${err.digest}` : 'no-digest';
  const raw = /timeout exceeded|Connection terminated/i.test(String(err?.message ?? e)) ? 'RAW-POOL-TEXT' : 'no-pool-text';
  return `THROW ${tag} ${raw} msg="${msg}"`;
}
