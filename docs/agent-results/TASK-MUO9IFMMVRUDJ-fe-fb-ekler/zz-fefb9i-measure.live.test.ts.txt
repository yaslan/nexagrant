/**
 * TASK-MUO122UPRPJQH (İNCELEME-77 B-1) — confirm against a store that is slow,
 * down, or dies half-way, and a row save that fails.
 *
 * Confirm used to copy the staging object, delete it and check the copy inside
 * the row's interactive transaction. A store slower than Prisma's 5 s ended the
 * transaction; the catch deleted the copy as well, so the file was nowhere, and
 * the "DB offline" branch answered `success: true` with a made-up id. Now the
 * store calls run outside any transaction and the staging object is removed
 * only after the row is saved, so every failure below must leave: no row, the
 * staging object as uploaded, an error — and a second confirm that succeeds.
 *
 * The store is the MinIO of tests/evidence-upload-s3.live.test.ts (same
 * EVIDENCE_S3_TEST_* variables, same skip in CI, where the job has no S3
 * service), reached through an in-process HTTP proxy that injects the faults.
 */

const SUITE = 'evidence-confirm-store-faults.live';
const S3 = {
  endpoint: process.env.EVIDENCE_S3_TEST_ENDPOINT,
  accessKey: process.env.EVIDENCE_S3_TEST_ACCESS_KEY,
  secretKey: process.env.EVIDENCE_S3_TEST_SECRET_KEY,
  bucket: process.env.EVIDENCE_S3_TEST_BUCKET,
};
const RLS_URL = envValue('DATABASE_URL_RLS_TEST');
const ADMIN_URL = envValue('DATABASE_URL');
const RUN = `evflt-${Date.now().toString(36)}`;

type Fault = 'pass' | 'down' | 'dieAfterCopy' | 'slowAfterCopy';

const state = vi.hoisted(() => {
  // storage.ts builds its client when imported: point it at the proxy first.
  const e = process.env;
  const proxyPort = 41000 + (process.pid % 15000);
  if (e.EVIDENCE_S3_TEST_ENDPOINT) {
    e.R2_ACCOUNT_ID = 'minio-test';
    e.R2_ENDPOINT = `http://127.0.0.1:${proxyPort}`;
    e.R2_ACCESS_KEY_ID = e.EVIDENCE_S3_TEST_ACCESS_KEY;
    e.R2_SECRET_ACCESS_KEY = e.EVIDENCE_S3_TEST_SECRET_KEY;
    e.R2_BUCKET = e.EVIDENCE_S3_TEST_BUCKET;
  }
  return {
    proxyPort,
    rls: undefined as PrismaClient | undefined,
    /** Confirm's own pool, built as src/lib/db.ts builds it (LOCK_POOL_OPTIONS). */
    rlsLock: undefined as PrismaClient | undefined,
    session: null as null | { user: { id: string }; tenantId: string; role: string },
    fault: 'pass' as Fault,
    copied: false,
    /** Runs while the proxy holds a CopyObject request, before it is forwarded. */
    beforeCopy: null as null | (() => Promise<void>),
    /** Makes one evidence call inside confirm's transaction fail, as a database error would. */
    dbFault: null as null | 'findFirst' | 'create',
    /** Holds confirm's process (not the database) around its "already recorded?" read. */
    dbPause: null as null | { before?: number; after?: number; afterLock?: number },
    /** Makes the database itself slow on confirm's "already recorded?" read (pg_sleep, seconds). */
    dbSleep: 0,
    /** The limits confirm's transaction ran its row save under. */
    seenLimits: null as null | Record<string, string>,
    /** Runs before the proxy forwards any request; may hold it. */
    hook: null as null | ((r: { method: string; path: string; isCopy: boolean; gone: Promise<void> }) => Promise<void>),
    requests: [] as string[],
  };
});

vi.mock('@/lib/auth', () => ({ auth: async () => state.session }));
vi.mock('@/lib/db', async (importOriginal) => {
  const actual = await importOriginal<typeof import('@/lib/db')>();
  // Only the named `evidence` call fails: everything else runs as usual.
  // `dbPause` / `dbSleep` slow the "already recorded?" read down; the row
  // save records the transaction's limits.
  type Raw = { $queryRaw: <R>(q: TemplateStringsArray, ...v: unknown[]) => Promise<R> };
  const pause = (ms?: number) => (ms ? new Promise((res) => setTimeout(res, ms)) : undefined);
  const failing = <T extends object>(tx: T): T =>
    new Proxy(tx, {
      get(target, prop, receiver) {
        const value = Reflect.get(target, prop, receiver);
        if (prop === '$executeRaw') {
          return async (strings: TemplateStringsArray, ...values: unknown[]) => {
            const done = await (value as (s: TemplateStringsArray, ...v: unknown[]) => Promise<number>).call(target, strings, ...values);
            if (strings.join('').includes('pg_advisory_xact_lock')) await pause(state.dbPause?.afterLock);
            return done;
          };
        }
        if (prop !== 'evidence') return value;
        return new Proxy(value as object, {
          get(d, m, r) {
            const real = Reflect.get(d, m, r) as (...a: unknown[]) => Promise<unknown>;
            if (m === state.dbFault) {
              return async () => {
                throw new Error(`injected: evidence.${String(m)} failed`);
              };
            }
            if (m === 'findFirst') {
              return async (...a: unknown[]) => {
                await pause(state.dbPause?.before);
                if (state.dbSleep) await (target as unknown as Raw).$queryRaw`SELECT pg_sleep(${state.dbSleep})::text`;
                const found = await real.apply(d, a);
                await pause(state.dbPause?.after);
                return found;
              };
            }
            if (m === 'create') {
              return async (...a: unknown[]) => {
                const [limits] = await (target as unknown as Raw).$queryRaw<Record<string, string>[]>`
                  SELECT current_setting('lock_timeout') AS lock_timeout,
                         current_setting('statement_timeout') AS statement_timeout,
                         current_setting('idle_in_transaction_session_timeout') AS idle_in_transaction_session_timeout`;
                state.seenLimits = limits ?? null;
                return real.apply(d, a);
              };
            }
            return typeof real === 'function' ? real.bind(d) : real;
          },
        });
      },
    });
  return {
    ...actual,
    withTenant: (tenantId: string, cb: Parameters<typeof actual.withTenant>[1], opts?: Parameters<typeof actual.withTenant>[2]) =>
      actual.withTenantOn(state.rls!, tenantId, (tx) => cb(failing(tx)), opts),
    // Confirm's own pool: a client of its own, with the production pool options.
    withTenantLockPool: (tenantId: string, cb: Parameters<typeof actual.withTenant>[1], opts?: Parameters<typeof actual.withTenant>[2]) =>
      actual.withTenantOn(state.rlsLock!, tenantId, (tx) => cb(failing(tx)), { maxWait: actual.LOCK_POOL_QUEUE_MS, ...opts }),
  };
});

const { createDbClient, withTenantOn, LOCK_POOL_MAX, LOCK_POOL_NAME, LOCK_POOL_OPTIONS } = await import('@/lib/db');

const evidenceIds: string[] = [];
const keys: string[] = [];

const finalOf = (stagingKey: string) => stagingKey.replace(/^([^/]+)\/incoming\//, '$1/');
const PDF = (text: string) => Buffer.from(`%PDF-1.4\n% ${text}\n%%EOF\n`);
const sleep = (ms: number) => new Promise((res) => setTimeout(res, ms));

const put = (url: string, body: Buffer) =>
  fetch(url, { method: 'PUT', headers: { 'Content-Type': 'application/pdf' }, body: new Uint8Array(body) });

async function uploaded(name: string) {
  const body = PDF(name);
  const p = await evidence.getPresignedUploadUrlAction('certificate', `${name}.pdf`, 'application/pdf', body.length);
  if (!p.success) throw new Error(`presign refused: ${p.error}`);
  keys.push(p.objectKey, finalOf(p.objectKey));
  expect((await put(p.uploadUrl, body)).status).toBe(200);
  return { ...p, finalKey: finalOf(p.objectKey), body };
}

const confirm = (objectKey: string) =>
  evidence.confirmEvidenceUploadAction({
    title: `${RUN} upload`,
    category: 'CAT-03',
    objectKey,
    fileName: 'belge.pdf',
    fileSizeBytes: 10,
    mimeType: 'application/pdf',
  });

const rowsOf = (finalKey: string) => admin!.evidence.findMany({ where: { tenantId: t.a, r2ObjectKey: finalKey } });

/** The store as it is, read with the faults off. */
async function stored(key: string) {
  const fault = state.fault;
  state.fault = 'pass';
  try {
    return await storage.getObjectBytes(key);
  } finally {
    state.fault = fault;
  }
}

/** A failed confirm: an error, no row, and the upload still under its staging key. */
async function expectNothingLost(up: Awaited<ReturnType<typeof uploaded>>, res: Awaited<ReturnType<typeof confirm>>, error: string) {
  expect(res).toEqual({ success: false, error });
  expect(await rowsOf(up.finalKey)).toHaveLength(0);
  expect((await stored(up.objectKey))?.equals(up.body), 'the staging object is kept for the next try').toBe(true);
}

/** The retry: one row with the store's figures, the checked bytes under the final key, no staging object. */
async function expectRetrySucceeds(up: Awaited<ReturnType<typeof uploaded>>) {
  state.fault = 'pass';
  state.dbFault = null;
  state.dbPause = null;
  state.dbSleep = 0;
  state.hook = null;
  const res = await confirm(up.objectKey);
  expect(res.success).toBe(true);
  const rows = await rowsOf(up.finalKey);
  evidenceIds.push(...rows.map((r) => r.id));
  expect(rows).toHaveLength(1);
  expect(res.success && res.evidence && 'id' in res.evidence ? res.evidence.id : null).toBe(rows[0]?.id);
  expect(rows[0]?.fileSizeBytes).toBe(up.body.length);
  expect((await stored(up.finalKey))?.equals(up.body)).toBe(true);
  expect(await stored(up.objectKey), 'the staging object goes once the row is saved').toBeNull();
}

beforeAll(async () => {
  if (!S3.endpoint || !S3.accessKey || !S3.secretKey || !S3.bucket) {
    skipReason = 'EVIDENCE_S3_TEST_* not set (no S3-compatible store to test against).';
    if (!isCi()) console.warn(`[${SUITE}] skipped: ${skipReason}`);
    return;
  }
  const rlsUrl = requireRlsTestUrl(RLS_URL, SUITE);
  if (!ADMIN_URL) skipOrFailInCi(SUITE, 'DATABASE_URL is not set');
  if (!rlsUrl || !ADMIN_URL) {
    skipReason = 'DATABASE_URL_RLS_TEST / DATABASE_URL not set. Run `pnpm db:rls-role`.';
    return;
  }

  const upstream = new URL(S3.endpoint);
  proxy = http.createServer(async (req, res) => {
    const isCopy = req.method === 'PUT' && Boolean(req.headers['x-amz-copy-source']);
    state.requests.push(`${req.method}${isCopy ? '(copy)' : ''} ${(req.url ?? '').split('?')[0]}`);
    if (state.fault === 'down' || (state.fault === 'dieAfterCopy' && state.copied)) {
      req.socket.destroy();
      return;
    }
    if (state.fault === 'slowAfterCopy' && state.copied) await sleep(6_000);
    if (isCopy && state.beforeCopy) {
      const hook = state.beforeCopy;
      state.beforeCopy = null;
      await hook();
    }
    if (state.hook) {
      // Settles when the client gives up on the request (or it is answered).
      const gone = new Promise<void>((resolve) => res.once('close', () => resolve()));
      await state.hook({ method: req.method ?? '', path: (req.url ?? '').split('?')[0] ?? '', isCopy, gone });
    }
    // A client that gave up while the request was held is not forwarded.
    if (req.socket.destroyed) return;
    const up = http.request(
      { host: upstream.hostname, port: upstream.port, method: req.method, path: req.url, headers: req.headers },
      (ur) => {
        res.writeHead(ur.statusCode ?? 502, ur.headers);
        ur.pipe(res);
        ur.on('end', () => {
          if (isCopy && (ur.statusCode ?? 0) < 300) state.copied = true;
        });
      }
    );
    up.on('error', () => res.destroy());
    req.pipe(up);
  });
  await new Promise<void>((resolve, reject) => {
    proxy!.once('error', reject);
    proxy!.listen(state.proxyPort, '127.0.0.1', () => resolve());
  });

  state.rls = createDbClient(rlsUrl);
  state.rlsLock = createDbClient(rlsUrl, LOCK_POOL_OPTIONS);
  admin = createDbClient(ADMIN_URL);
  try {
    await state.rls.$connect();
  } catch (e) {
    skipReason = `database unavailable: ${e instanceof Error ? e.message : String(e)}`;
    skipOrFailInCi(SUITE, skipReason);
    return;
  }
  await assertRlsSubjectRole(state.rls, SUITE);
  expect(storage.hasObjectStore, 'storage.ts must be talking to the test store').toBe(true);

  const tenant = await admin.tenant.create({
    data: { name: `${RUN} Kurum`, slug: RUN, country: 'TR', kind: 'ORGANIZATION' },
    select: { id: true },
  });
  t.a = tenant.id;
  t.b = (
    await admin.tenant.create({
      data: { name: `${RUN} Diger Kurum`, slug: `${RUN}-b`, country: 'TR', kind: 'ORGANIZATION' },
      select: { id: true },
    })
  ).id;
  t.userA = (await admin.user.create({ data: { email: `${RUN}@example.invalid`, name: 'a' } })).id;
  ready = true;
});

beforeEach(({ skip }) => {
  if (!ready) skip(skipReason);
  state.session = { user: { id: t.userA }, tenantId: t.a, role: 'TENANT_OWNER' };
  state.fault = 'pass';
  state.copied = false;
  state.beforeCopy = null;
  state.dbFault = null;
  state.dbPause = null;
  state.dbSleep = 0;
  state.seenLimits = null;
  state.hook = null;
  state.requests = [];
});

afterAll(async () => {
  state.fault = 'pass';
  for (const key of keys) await storage.deleteObjectBytes(key).catch(() => {});
  if (admin) {
    await admin.evidence.deleteMany({ where: { id: { in: presentIds(evidenceIds) } } }).catch(() => {});
    for (const tenantId of presentIds([t.a, t.b])) {
      await admin.evidence.deleteMany({ where: { tenantId: requireId(tenantId, 'tenantId') } }).catch(() => {});
      await admin.tenant.delete({ where: { id: tenantId } }).catch(() => {});
    }
    await admin.user.deleteMany({ where: { id: { in: presentIds([t.userA]) } } }).catch(() => {});
    await admin.$disconnect();
  }
  await state.rls?.$disconnect();
  await state.rlsLock?.$disconnect();
  await new Promise((resolve) => (proxy ? proxy.close(resolve) : resolve(undefined)));
});

describe('B-1 (TASK-MUO122UPRPJQH): a slow or failing store never loses the upload', () => {
  test('crw-fb’s case: every store call after the copy takes 6 s — the confirm is slow, and it succeeds', async () => {
    const up = await uploaded('yavas');
    state.fault = 'slowAfterCopy';
    const res = await confirm(up.objectKey);
    expect(res.success, JSON.stringify(res)).toBe(true);
    const rows = await rowsOf(up.finalKey);
    evidenceIds.push(...rows.map((r) => r.id));
    expect(rows).toHaveLength(1);
    expect(res.success && res.evidence && 'id' in res.evidence ? res.evidence.id : null).toBe(rows[0]?.id);
    expect((await stored(up.finalKey))?.equals(up.body)).toBe(true);
    expect(await stored(up.objectKey)).toBeNull();
  }, 90_000);

  test('the row save fails after the copy: save_failed, no row, the upload kept; the retry succeeds', async () => {
    const up = await uploaded('satir-hatasi');
    state.dbFault = 'create';
    const res = await confirm(up.objectKey);
    expect(state.copied, 'the copy did go through').toBe(true);
    await expectNothingLost(up, res, 'save_failed');
    await expectRetrySucceeds(up);
  }, 60_000);

  test('T-2: the "already recorded?" read fails: save_failed before any copy, the upload kept; the retry succeeds', async () => {
    const up = await uploaded('on-kontrol');
    state.dbFault = 'findFirst';
    const res = await confirm(up.objectKey);
    await expectNothingLost(up, res, 'save_failed');
    expect(state.requests.some((r) => r.startsWith('PUT(copy)')), 'nothing is copied without the check').toBe(false);
    await expectRetrySucceeds(up);
  }, 60_000);

  test('the store holds the copy past its deadline: storage_unavailable in bounded time, nothing lost; the retry succeeds', async () => {
    const up = await uploaded('asiri-yavas');
    let held = false;
    state.hook = async ({ isCopy }) => {
      if (!isCopy || held) return;
      held = true;
      await sleep(storage.EVIDENCE_COPY_DEADLINE_MS + 3_000);
    };
    const started = Date.now();
    const res = await confirm(up.objectKey);
    const took = Date.now() - started;
    expect(took, `gave up after ${took} ms`).toBeGreaterThanOrEqual(storage.EVIDENCE_COPY_DEADLINE_MS - 1_000);
    expect(took).toBeLessThan(storage.EVIDENCE_COPY_DEADLINE_MS + 2_500);
    await expectNothingLost(up, res, 'storage_unavailable');
    expect(await stored(up.finalKey), 'the abandoned copy never reached the store').toBeNull();
    await sleep(3_500);
    await expectRetrySucceeds(up);
  }, 90_000);

  test('the store is down before the copy: storage_unavailable, nothing lost; the retry succeeds', async () => {
    const up = await uploaded('kapali');
    state.fault = 'down';
    const res = await confirm(up.objectKey);
    await expectNothingLost(up, res, 'storage_unavailable');
    expect(state.requests.some((r) => r.startsWith('PUT(copy)')), 'no copy was attempted').toBe(false);
    await expectRetrySucceeds(up);
  }, 60_000);

  test('the store dies right after the copy: storage_unavailable, nothing lost; the retry succeeds', async () => {
    const up = await uploaded('kopya-sonrasi');
    state.fault = 'dieAfterCopy';
    const res = await confirm(up.objectKey);
    expect(state.copied, 'the copy did go through').toBe(true);
    await expectNothingLost(up, res, 'storage_unavailable');
    await expectRetrySucceeds(up);
  }, 60_000);

  test('the staging object is replaced between the check and the copy: the unchecked bytes never reach the final key', async () => {
    const up = await uploaded('degisen');
    const html = Buffer.from('<html>'.padEnd(up.body.length - 7, 'x') + '</html>');
    expect(html.length).toBe(up.body.length);
    // Same signed URL, type and length: the store takes it, mid-confirm.
    state.beforeCopy = async () => {
      expect((await put(up.uploadUrl, html)).status).toBe(200);
    };
    const res = await confirm(up.objectKey);
    expect(res).toEqual({ success: false, error: 'storage_unavailable' });
    expect(await rowsOf(up.finalKey)).toHaveLength(0);
    expect(await stored(up.finalKey), 'the copy was refused, the final key holds nothing').toBeNull();

    // The next confirm checks what is there now, and refuses it for good.
    expect(await confirm(up.objectKey)).toEqual({ success: false, error: 'unsupported_type' });
    expect(await stored(up.objectKey)).toBeNull();
    expect(await stored(up.finalKey)).toBeNull();
    expect(await rowsOf(up.finalKey)).toHaveLength(0);
  }, 60_000);

  test('positive control: through the same proxy with no fault, one confirm is one row and no staging object', async () => {
    const up = await uploaded('temiz');
    await expectRetrySucceeds(up);
    expect(state.requests.filter((r) => r.startsWith('PUT(copy)'))).toHaveLength(1);
  });
});

describe('Y-1 (TASK-MUO64MVW8TLMX, İNCELEME-82): overlapping confirms of one key', () => {
  /** Same type and length as `up.body`, other bytes: what the still-valid signed PUT can put there. */
  const other = (up: Awaited<ReturnType<typeof uploaded>>) => {
    const v2 = PDF('X'.repeat(up.body.length - PDF('').length));
    expect(v2.length).toBe(up.body.length);
    expect(v2.equals(up.body)).toBe(false);
    return v2;
  };

  test('RACE-2: a second confirm that starts before the first one saved its row cannot change the file the first one recorded', async () => {
    const up = await uploaded('yaris-2');
    const v2 = other(up);
    let copies = 0;
    let second: ReturnType<typeof confirm> | undefined;
    state.hook = async ({ method, path, isCopy }) => {
      if (isCopy) {
        copies += 1;
        // crw-fb's timing: the second copy is slow enough to land after the first row is committed.
        if (copies === 2) await sleep(2_500);
        return;
      }
      // The first confirm reads its copy back: v1 is on the final key, its row is not saved yet.
      if (method === 'HEAD' && path.endsWith(`/${up.finalKey}`) && copies === 1 && !second) {
        expect((await put(up.uploadUrl, v2)).status).toBe(200);
        second = confirm(up.objectKey);
        await sleep(1_500);
      }
    };

    const first = await confirm(up.objectKey);
    expect(second, 'the second confirm was started while the first one ran').toBeDefined();
    const res2 = await second!;
    state.hook = null;

    expect(first.success, JSON.stringify(first)).toBe(true);
    expect(res2).toEqual({ success: false, error: 'already_confirmed' });
    const rows = await rowsOf(up.finalKey);
    evidenceIds.push(...rows.map((r) => r.id));
    expect(rows).toHaveLength(1);
    expect(first.success && first.evidence && 'id' in first.evidence ? first.evidence.id : null).toBe(rows[0]?.id);
    expect((await stored(up.finalKey))?.equals(up.body), 'the recorded file is still the one the first confirm checked').toBe(true);
    expect(copies, 'the second confirm copied nothing').toBe(1);
    expect(await stored(up.objectKey)).toBeNull();
  }, 60_000);

  test('a confirm waits for the one ahead of it only so long: save_failed, nothing copied or lost; the first still succeeds', async () => {
    const up = await uploaded('kilit-bekleme');
    let copies = 0;
    let second: ReturnType<typeof confirm> | undefined;
    let secondTook = 0;
    state.hook = async ({ isCopy }) => {
      if (!isCopy) return;
      copies += 1;
      if (copies !== 1) return;
      // The first confirm holds the lock through a copy slower than the second one's wait.
      const started = Date.now();
      second = confirm(up.objectKey).then((r) => {
        secondTook = Date.now() - started;
        return r;
      });
      await sleep(11_000);
    };

    const first = await confirm(up.objectKey);
    const res2 = await second!;
    state.hook = null;

    expect(res2).toEqual({ success: false, error: 'save_failed' });
    expect(secondTook, `the second confirm answered after ${secondTook} ms`).toBeLessThan(11_000);
    // The whole lock_timeout, not the shorter statement_timeout (TASK-MUO9IFMMVRUDJ).
    expect(secondTook).toBeGreaterThanOrEqual(9_000);
    expect(first.success, JSON.stringify(first)).toBe(true);
    const rows = await rowsOf(up.finalKey);
    evidenceIds.push(...rows.map((r) => r.id));
    expect(rows).toHaveLength(1);
    expect(copies).toBe(1);
    expect((await stored(up.finalKey))?.equals(up.body)).toBe(true);
    // Sent again, it finds the row.
    expect(await confirm(up.objectKey)).toEqual({ success: false, error: 'already_confirmed' });
  }, 60_000);

  test('five confirms at once: one row, one copy, the uploaded bytes on the final key', async () => {
    const up = await uploaded('bes-birden');
    const results = await Promise.all(Array.from({ length: 5 }, () => confirm(up.objectKey)));
    const ok = results.filter((r) => r.success);
    expect(ok).toHaveLength(1);
    for (const r of results.filter((x) => !x.success)) expect(r).toEqual({ success: false, error: 'already_confirmed' });
    const rows = await rowsOf(up.finalKey);
    evidenceIds.push(...rows.map((r) => r.id));
    expect(rows).toHaveLength(1);
    expect(state.requests.filter((r) => r.startsWith('PUT(copy)'))).toHaveLength(1);
    expect((await stored(up.finalKey))?.equals(up.body)).toBe(true);
    expect(await stored(up.objectKey)).toBeNull();
  }, 60_000);
});

describe('İNCELEME-92 (TASK-MUO9IFMMVRUDJ): the lock held across the copy costs nobody else', () => {
  /** Same derivation as confirmLockKey in evidence/actions.ts. */
  const lockKeyOf = (finalKey: string) =>
    createHash('sha256').update(`evidence-confirm:${finalKey}`).digest().readBigInt64BE(0);
  const lockHeld = async (finalKey: string) =>
    Number(
      (
        await admin!.$queryRaw<{ n: bigint }[]>`
          SELECT count(*)::bigint AS n FROM pg_locks
          WHERE locktype = 'advisory' AND granted AND ((classid::bigint << 32) | objid::bigint) = ${lockKeyOf(finalKey)}`
      )[0]?.n ?? 0
    );
  const lockPoolConnections = async () =>
    Number(
      (
        await admin!.$queryRaw<{ n: bigint }[]>`
          SELECT count(*)::bigint AS n FROM pg_stat_activity WHERE application_name = ${LOCK_POOL_NAME}`
      )[0]?.n ?? 0
    );
  /** Watches the lock every 100 ms until `until` settles; the time it was first seen gone, or null if it was held throughout. */
  const watchLock = (finalKey: string, until: Promise<unknown>) => {
    let done = false;
    void until.finally(() => (done = true));
    return (async () => {
      while (!done) {
        if ((await lockHeld(finalKey)) === 0) return Date.now();
        await sleep(100);
      }
      return null;
    })();
  };

  test('B-1: ten confirms held by a slow store do not take the application pool — another tenant’s queries run as usual', async () => {
    const ups = await Promise.all(Array.from({ length: 10 }, (_, i) => uploaded(`havuz-${i}`)));
    state.hook = async ({ isCopy }) => {
      if (isCopy) await sleep(8_000);
    };
    const all = ups.map((u) => confirm(u.objectKey));
    await sleep(1_500);

    const probes: { ok: boolean; ms: number }[] = [];
    let peakLockConnections = 0;
    for (let i = 0; i < 4; i += 1) {
      const t0 = Date.now();
      // Tenant B, on the application pool — the same 10-connection pg.Pool as src/lib/db.ts.
      const ok = await withTenantOn(state.rls!, t.b, (tx) => tx.evidence.count())
        .then(() => true)
        .catch(() => false);
      probes.push({ ok, ms: Date.now() - t0 });
      peakLockConnections = Math.max(peakLockConnections, await lockPoolConnections());
      await sleep(1_000);
    }
    const res = await Promise.all(all);
    state.hook = null;
    for (const u of ups) evidenceIds.push(...(await rowsOf(u.finalKey)).map((r) => r.id));
    console.info(
      `[B-1] confirms=${res.map((r) => (r.success ? 'ok' : r.error)).join(',')} probes=${probes.map((p) => `${p.ok ? 'ok' : 'FAIL'} ${p.ms}ms`).join(' | ')} lockPoolConnections(peak)=${peakLockConnections}`
    );

    for (const p of probes) {
      expect(p.ok, 'tenant B’s query got a connection').toBe(true);
      expect(p.ms).toBeLessThan(500);
    }
    expect(peakLockConnections).toBeGreaterThan(0);
    expect(peakLockConnections).toBeLessThanOrEqual(LOCK_POOL_MAX);
    // The pool holds LOCK_POOL_MAX confirms; the rest waited their turn or gave up — retryably, nothing lost.
    const saved = res.filter((r) => r.success);
    expect(saved.length).toBeGreaterThanOrEqual(LOCK_POOL_MAX);
    for (const [i, r] of res.entries()) {
      if (r.success) continue;
      await expectNothingLost(ups[i]!, r, 'save_failed');
      await expectRetrySucceeds(ups[i]!);
    }
  }, 120_000);

  test('T-4: two uploads of one tenant do not wait for each other', async () => {
    const [slow, fast] = await Promise.all([uploaded('t4-yavas'), uploaded('t4-hizli')]);
    let fastTook = 0;
    let fastRes: Awaited<ReturnType<typeof confirm>> | undefined;
    state.hook = async ({ isCopy, path }) => {
      if (!isCopy || !path.endsWith(`/${slow.finalKey}`)) return;
      const t0 = Date.now();
      fastRes = await confirm(fast.objectKey);
      fastTook = Date.now() - t0;
      await sleep(Math.max(0, 6_000 - fastTook));
    };
    const slowRes = await confirm(slow.objectKey);
    state.hook = null;
    evidenceIds.push(...(await rowsOf(slow.finalKey)).map((r) => r.id), ...(await rowsOf(fast.finalKey)).map((r) => r.id));
    expect(slowRes.success).toBe(true);
    expect(fastRes?.success, JSON.stringify(fastRes)).toBe(true);
    expect(fastTook, `the second upload answered after ${fastTook} ms`).toBeLessThan(2_000);
  }, 60_000);

  test('T-3: the limits are set inside confirm’s transaction and never left on a pooled connection', async () => {
    const ups = await Promise.all(Array.from({ length: LOCK_POOL_MAX }, (_, i) => uploaded(`t3-${i}`)));
    const res = await Promise.all(ups.map((u) => confirm(u.objectKey)));
    for (const u of ups) evidenceIds.push(...(await rowsOf(u.finalKey)).map((r) => r.id));
    expect(res.every((r) => r.success)).toBe(true);
    expect(state.seenLimits).toEqual({
      lock_timeout: '10s',
      statement_timeout: '5s',
      idle_in_transaction_session_timeout: '20s',
    });

    // Every connection of the pool at once (pg_sleep keeps each one busy), outside any transaction.
    const after = await Promise.all(
      Array.from({ length: LOCK_POOL_MAX }, () =>
        state.rlsLock!.$queryRaw<{ pid: number; l: string; s: string; i: string }[]>`
          SELECT pg_backend_pid() AS pid, current_setting('lock_timeout') AS l, current_setting('statement_timeout') AS s,
                 current_setting('idle_in_transaction_session_timeout') AS i, pg_sleep(0.3)::text AS z`
      )
    );
    expect(new Set(after.map((r) => r[0]!.pid)).size).toBe(LOCK_POOL_MAX);
    for (const r of after) expect({ l: r[0]!.l, s: r[0]!.s, i: r[0]!.i }).toEqual({ l: '0', s: '0', i: '0' });
  }, 60_000);

  test('Y-2: a process that freezes inside confirm’s transaction loses the lock after 20 s, not hours; the retry succeeds', async () => {
    const up = await uploaded('donan');
    // The process stops between the lock and its next statement: the connection sits idle in the transaction.
    state.dbPause = { before: 24_000 };
    const started = Date.now();
    const running = confirm(up.objectKey);
    await sleep(1_000);
    expect(await lockHeld(up.finalKey), 'the frozen confirm holds the lock').toBe(1);
    const releasedAt = await watchLock(up.finalKey, running);
    const res = await running;
    const releasedAfter = releasedAt === null ? null : releasedAt - started;
    console.info(`[Y-2] lock released after ${releasedAfter} ms; confirm=${JSON.stringify(res)}`);

    expect(releasedAfter, 'released by idle_in_transaction_session_timeout').not.toBeNull();
    expect(releasedAfter!).toBeGreaterThanOrEqual(19_000);
    expect(releasedAfter!).toBeLessThan(23_000);
    await expectNothingLost(up, res, 'save_failed');
    expect(state.requests.some((r) => r.startsWith('PUT(copy)'))).toBe(false);
    // The pool dropped the closed connection: the next confirm gets a working one.
    await expectRetrySucceeds(up);
  }, 90_000);

  test('Y-3: the copy ends before a connection idle through it could be closed — the lock is never gone while the copy runs', async () => {
    const up = await uploaded('idle-pay');
    // 12 s pass after the read: 20 − 12 − 2 leaves the copy about 6 s, not 15.
    state.dbPause = { after: 12_000 };
    let copyGoneAt = 0;
    state.hook = async ({ isCopy, gone }) => {
      if (!isCopy) return;
      await gone;
      copyGoneAt = Date.now();
    };
    const started = Date.now();
    const running = confirm(up.objectKey);
    await sleep(1_000);
    const releasedAt = await watchLock(up.finalKey, running);
    const res = await running;
    state.hook = null;
    console.info(
      `[Y-3 idle] copy given up after ${copyGoneAt - started} ms; lock ${releasedAt === null ? 'held until confirm answered' : `gone after ${releasedAt - started} ms`}`
    );

    expect(copyGoneAt, 'the copy was attempted and given up').toBeGreaterThan(0);
    expect(copyGoneAt - started).toBeLessThan(19_000);
    // Held until confirm answered, or at least until the copy was given up.
    if (releasedAt !== null) expect(releasedAt, 'the lock outlived the copy').toBeGreaterThanOrEqual(copyGoneAt);
    await expectNothingLost(up, res, 'storage_unavailable');
    expect(await stored(up.finalKey)).toBeNull();
    await expectRetrySucceeds(up);
  }, 90_000);

  test('Y-3: the copy ends before the transaction’s own timeout — a late lock and a stalled process leave it less time', async () => {
    const up = await uploaded('tx-pay');
    // Another confirm of the key holds the lock for 9 s.
    const holder = admin!.$transaction(
      async (tx) => {
        await tx.$executeRaw`SELECT pg_advisory_xact_lock(${lockKeyOf(up.finalKey)})`;
        await sleep(9_000);
      },
      { timeout: 30_000 }
    );
    await sleep(300);
    // Then the process stalls 15 s after taking the lock, and the read takes 4.5 s:
    // about 29 s of the transaction's 40 are gone, ~4 s are left for the copy instead of ~13.
    state.dbPause = { afterLock: 15_000 };
    state.dbSleep = 4.5;
    let copyGoneAt = 0;
    state.hook = async ({ isCopy, gone }) => {
      if (!isCopy) return;
      await gone;
      copyGoneAt = Date.now();
    };
    const started = Date.now();
    const running = confirm(up.objectKey);
    await holder;
    await sleep(1_000);
    const releasedAt = await watchLock(up.finalKey, running);
    const res = await running;
    state.hook = null;
    console.info(
      `[Y-3 tx] copy given up after ${copyGoneAt - started} ms; lock ${releasedAt === null ? 'held until confirm answered' : `gone after ${releasedAt - started} ms`}`
    );

    expect(copyGoneAt, 'the copy was attempted and given up').toBeGreaterThan(0);
    expect(copyGoneAt - started).toBeLessThan(38_000);
    if (releasedAt !== null) expect(releasedAt, 'the lock outlived the copy').toBeGreaterThanOrEqual(copyGoneAt);
    await expectNothingLost(up, res, 'storage_unavailable');
    await expectRetrySucceeds(up);
  }, 90_000);

  test('Y-3: a slow database after the lock is cut off by statement_timeout — save_failed in ~5 s, nothing copied', async () => {
    const up = await uploaded('yavas-db');
    state.dbSleep = 8;
    const started = Date.now();
    const res = await confirm(up.objectKey);
    const took = Date.now() - started;
    expect(took, `answered after ${took} ms`).toBeLessThan(7_000);
    await expectNothingLost(up, res, 'save_failed');
    expect(state.requests.some((r) => r.startsWith('PUT(copy)'))).toBe(false);
    expect(await lockHeld(up.finalKey)).toBe(0);
    await expectRetrySucceeds(up);
  }, 60_000);

  test.each(['HEAD', 'GET'])('T-5: a store that holds the %s of the staging object is given up on within the deadline', async (held) => {
    const up = await uploaded(`t5-${held.toLowerCase()}`);
    state.hook = async ({ method, path, isCopy }) => {
      if (!isCopy && method === held && path.endsWith(`/${up.objectKey}`)) await sleep(4_000);
    };
    const started = Date.now();
    const res = await storage.copyIncomingObject(up.objectKey, up.finalKey, 1_500);
    const took = Date.now() - started;
    state.hook = null;
    expect(res).toEqual({ ok: false, error: 'storage_unavailable' });
    expect(took, `gave up after ${took} ms`).toBeLessThan(3_000);
    expect(state.requests.some((r) => r.startsWith('PUT(copy)'))).toBe(false);
    await sleep(3_000);
    expect(await stored(up.finalKey)).toBeNull();
    await expectRetrySucceeds(up);
  }, 60_000);
});

describe('MEASURE', () => {
  test('MinIO paused 10 s while tenant A confirms 10 uploads; tenant B queries + pg_stat_activity every 500 ms', async () => {
    const { execSync } = await import('node:child_process');
    const ups = await Promise.all(Array.from({ length: 10 }, (_, i) => uploaded(`olcum-${i}`)));
    execSync('docker pause fefb9i-minio');
    const pausedAt = Date.now();
    const unpause = setTimeout(() => execSync('docker unpause fefb9i-minio'), 10_000);
    const t0 = Date.now();
    const all = ups.map((u) => confirm(u.objectKey).then((r) => ({ r, ms: Date.now() - t0 })));
    const rows: string[] = ['| t (sn) | B sorgusu | B süre (ms) | kilit havuzu bağlantı (active / idle in tx / idle) | uygulama havuzu fabrika_app bağlantı (active / idle in tx / idle) |', '|---|---|---|---|---|'];
    for (let i = 0; i < 18; i += 1) {
      await sleep(500);
      const s = Date.now();
      const ok = await withTenantOn(state.rls!, t.b, (tx) => tx.evidence.count()).then(() => 'ok').catch((e) => `FAIL (${String(e?.message ?? e).split('\n').pop()?.slice(0, 50)})`);
      const ms = Date.now() - s;
      const act = await admin!.$queryRaw<{ app: string; state: string; n: bigint }[]>`
        SELECT coalesce(application_name,'') AS app, coalesce(state,'') AS state, count(*)::bigint AS n
        FROM pg_stat_activity WHERE usename = 'fabrika_app' GROUP BY 1,2`;
      const pick = (lock: boolean, st: string) => Number(act.filter((a) => (a.app === LOCK_POOL_NAME) === lock && a.state === st).reduce((m, a) => m + a.n, 0n));
      rows.push(`| ${((Date.now() - pausedAt) / 1000).toFixed(1)} | ${ok} | ${ms} | ${pick(true, 'active')} / ${pick(true, 'idle in transaction')} / ${pick(true, 'idle')} | ${pick(false, 'active')} / ${pick(false, 'idle in transaction')} / ${pick(false, 'idle')} |`);
    }
    const res = await Promise.all(all);
    clearTimeout(unpause);
    try { execSync('docker unpause fefb9i-minio 2>/dev/null'); } catch {}
    for (const u of ups) evidenceIds.push(...(await rowsOf(u.finalKey)).map((r) => r.id));
    console.info(['[MEASURE]', ...rows, `confirms: ${res.map((x) => `${x.r.success ? 'ok' : x.r.error}@${x.ms}ms`).join(', ')}`].join('\n'));
    // Retry the failed ones: nothing was lost.
    for (const [i, x] of res.entries()) if (!x.r.success) await expectRetrySucceeds(ups[i]!);
    console.info(`[MEASURE] retries: all ${res.filter((x) => !x.r.success).length} succeeded`);
  }, 180_000);
});
