async function uploaded(name: string, s: Session = sessA()) {
  return als.run(s, async () => {
    const body = PDF(name);
    const p = await evidence.getPresignedUploadUrlAction('certificate', `${name}.pdf`, 'application/pdf', body.length);
    if (!p.success) throw new Error(`presign refused: ${p.error}`);
    keys.push(p.objectKey, finalOf(p.objectKey));
    expect((await put(p.uploadUrl, body)).status).toBe(200);
    return { ...p, finalKey: finalOf(p.objectKey), body };
  });
}

async function stored(key: string) {
  const h = state.hook;
  state.hook = null;
  try {
    return await storage.getObjectBytes(key);
  } finally {
    state.hook = h;
  }
}
/** Is the confirm lock of this final key held by someone? (admin session, try-lock in its own tx) */
async function lockHeld(finalKey: string) {
  return admin.$transaction(async (tx) => {
    const [r] = await tx.$queryRaw<{ got: boolean }[]>`SELECT pg_try_advisory_xact_lock(${lockKeyOf(finalKey)}) AS got`;
    return !r!.got;
  });
}
type Act = { app: string; state: string; n: number };
async function activity(): Promise<Act[]> {
  return admin.$queryRaw<Act[]>`SELECT coalesce(application_name,'') AS app, coalesce(state,'') AS state, count(*)::int AS n
    FROM pg_stat_activity WHERE usename = 'fabrika_app' AND pid <> pg_backend_pid() GROUP BY 1, 2 ORDER BY 1, 2`;
}
const fmtAct = (a: Act[]) => a.map((r) => `${r.app || 'app'}:${r.state}=${r.n}`).join(' ');

        out.push(`${sig}: holder=${holder?.app}/${holder?.state};${confirmWhileHeld} lock released ${releasedMs} ms after signal; retry=${r.success ? 'success' : JSON.stringify(r)} rows=${rows.length} final==uploaded=${(await stored(up.finalKey))?.equals(up.body)}`);
        expect(r.success).toBe(true);
        expect(rows).toHaveLength(1);
        return releasedMs;
      } finally {
        try { process.kill(child.pid!, 'SIGKILL'); } catch { /* gone */ }
      }
    };
    const killed = await run('SIGKILL');
    const stopped = await run('SIGSTOP');
    for (const l of out) log(`D6 ${l}`);
    expect(killed).toBeLessThan(2_000);
    expect(stopped).toBeLessThan(25_000);
  }, 180_000);
