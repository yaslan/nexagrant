const { chromium } = require('@playwright/test');
const fs = require('fs');
(async () => {
  const [jar, raw, dir] = process.argv.slice(2);
  const line = fs.readFileSync(jar, 'utf8').split('\n').find((l) => l.includes('authjs.session-token'));
  const b = await chromium.launch({ channel: 'chrome' });
  const ctx = await b.newContext({ viewport: { width: 1280, height: 800 } });
  await ctx.addCookies([{ name: 'authjs.session-token', value: line.split('\t')[6], domain: 'localhost', path: '/' }]);
  const p = await ctx.newPage();
  const B = 'http://localhost:3491';
  const sess = async () => JSON.stringify(await (await p.request.get(B + '/api/auth/session')).json(), ['tenantId', 'role']);
  const text = async () => (await p.innerText('body')).slice(0, 260).replace(/\s+/g, ' ');
  let r = await p.goto(B + '/tr/default/dashboard', { waitUntil: 'load', timeout: 120000 });
  console.log('[B1] tenantless /tr/default/dashboard ->', r.status(), p.url(), '| session', await sess());
  console.log('     text:', await text());
  await p.screenshot({ path: dir + '/shot-1-tenantless-no-workspace.png' });
  r = await p.goto(B + '/tr/accept-invite?token=' + raw, { waitUntil: 'load', timeout: 120000 });
  await p.getByTestId('accept-invite-submit').waitFor({ timeout: 60000 });
  const dlg = p.getByRole('alertdialog');
  if (await dlg.waitFor({ timeout: 20000 }).then(() => true, () => false)) {
    const names = await dlg.getByRole('button').allInnerTexts();
    console.log('     cookie dialog buttons:', JSON.stringify(names));
    await dlg.getByRole('button').first().click();
    await dlg.waitFor({ state: 'hidden' });
  }
  console.log('[B2] accept screen ->', p.url().replace(raw, '<token>'), '| token in URL?', p.url().includes(raw));
  await p.screenshot({ path: dir + '/shot-2-accept-screen.png' });
  await Promise.all([p.waitForURL('**/tr/default/dashboard', { timeout: 120000 }), p.getByTestId('accept-invite-submit').click()]);
  await p.waitForLoadState('load');
  await p.waitForTimeout(1500);
  console.log('[B3] after "Kabul Et" ->', p.url(), '| session', await sess());
  console.log('     text:', await text());
  await p.screenshot({ path: dir + '/shot-3-after-accept-dashboard-C.png' });
  await p.reload({ waitUntil: 'load' });
  console.log('[B4] reload, same browser  | session', await sess());
  fs.writeFileSync(dir + '/j-after.txt', JSON.stringify(await ctx.cookies()));
  await b.close();
})().catch((e) => { console.error('FAIL', e.message); process.exit(1); });
