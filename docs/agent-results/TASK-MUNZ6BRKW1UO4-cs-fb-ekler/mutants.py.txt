def run(spec):
    out = []
    for part in spec.split('|'):
        if part == 'tsc': r = subprocess.run(['npx','tsc','--noEmit','-p','tsconfig.json'], cwd=W, capture_output=True, text=True)
        elif part == 'eslint': r = subprocess.run(['npx','eslint','src/app/api/cagrilar'], cwd=W, capture_output=True, text=True)
        else: r = subprocess.run(['npx','vitest','run', part], cwd=W, capture_output=True, text=True)
        out.append((part, r.returncode))
    return out
for mid, desc, spec, edits in M:
    if only and mid not in only: continue
    created = []
    try:
        for path, old, new in edits:
            p = W / path
            if old is None and not new.startswith('APPEND'):
                p.write_text(new); created.append(p); continue
            s = p.read_text()
            if old is None: p.write_text(s + new[len('APPEND'):]); continue
            assert s.count(old) == 1, f'{mid}: anchor not unique/absent in {path}: {old[:40]}'
            p.write_text(s.replace(old, new))
        res = run(spec)
        red = [f'{k}={c}' for k, c in res]
        verdict = 'ÖLDÜ (kırmızı)' if any(c != 0 for _, c in res) else 'YAŞADI (yeşil)'
        print(f'{mid} {desc}: {verdict}  [{", ".join(red)}]', flush=True)
    finally:
        for p in created: p.unlink()
        subprocess.run(['git','checkout','--','src'], cwd=W)
