// Also reads every server-action response body off the wire (RSC payload) and scans it.
import { createRequire } from 'node:module';
import { execFileSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
const SP = '/private/tmp/claude-501/-Users-macbook-alan/6c39b01f-1838-45a1-9335-1996d802b2b0/scratchpad';
const require = createRequire(`${SP}/wt94/package.json`);
const { chromium } = require('@playwright/test');
const { encode } = require('next-auth/jwt');
const B = 'http://127.0.0.1:47994';
const psql = (sql) => execFileSync('docker', ['exec', 'crwfb94-pg', 'psql', '-U', 'fabrika', '-d', 'fabrika', '-v', 'ON_ERROR_STOP=1', '-tAc', sql]).toString().trim();
const LEAK = /crw94|hunter2|tablo=|P[0-9]{4}\b|23503|23505|prisma|invocation|constraint|violat|\{detail\}/i;
const tr = JSON.parse(readFileSync(`${SP}/wt94/messages/tr.json`, 'utf8')).adminPlans.errors;
const en = JSON.parse(readFileSync(`${SP}/wt94/messages/en.json`, 'utf8')).adminPlans.errors;
const salt = 'authjs.session-token';
const cookieFor = (id, email) => encode({ secret: readFileSync(`${SP}/authsecret`, 'utf8').trim(), salt, maxAge: 3600, token: { sub: id, id, email } });

const browser = await chromium.launch({ executablePath: '/Users/macbook/Library/Caches/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-mac-arm64/chrome-headless-shell' });
let fails = 0;
const check = (label, ok, extra = '') => { if (!ok) fails++; console.log(`${ok ? 'OK  ' : 'FAIL'} ${label}${extra ? ` — ${extra}` : ''}`); };
async function ctxFor(id, email) {
  const ctx = await browser.newContext();
  await ctx.addCookies([{ name: salt, value: await cookieFor(id, email), url: B }]);
  await ctx.addInitScript(() => localStorage.setItem('fabrika_cookie_consent', JSON.stringify({ necessary: true, analytics: true, marketing: true })));
  return ctx;
}
const admin = await ctxFor('demo-user-superadmin', 'admin@yasinaslan.com.tr');
const page = await admin.newPage();
const wire = [];
page.on('response', async (r) => { if (r.request().method() === 'POST' && r.request().headers()['next-action']) wire.push(await r.text().catch(() => '')); });

const fault = (table, ops, errcode) => ({
  install: `CREATE OR REPLACE FUNCTION crw94_boom() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN RAISE EXCEPTION 'crw94-internal tablo=% secret=hunter2', TG_TABLE_NAME${errcode ? ` USING ERRCODE = '${errcode}'` : ''}; END $$;
            CREATE TRIGGER crw94_boom BEFORE ${ops} ON "${table}" FOR EACH ROW EXECUTE FUNCTION crw94_boom();`,
  remove: `DROP TRIGGER crw94_boom ON "${table}"; DROP FUNCTION crw94_boom();`,
});

async function scenario(locale, name, f, plan, act, noticeId, expectText, shot) {
  psql(f.install);
  wire.length = 0;
  try {
    await page.goto(`${B}/${locale}/admin/billing/plans`);
    if (plan) await page.getByTestId(`plan-row-${plan}`).click();
    await act();
    const notice = page.getByTestId(noticeId);
    await notice.waitFor({ timeout: 15000 });
    const text = (await notice.innerText()).trim();
    console.log(`-- [${locale}] ${name}: ${noticeId} = "${text}"`);
    check(`[${locale}] ${name}: expected sentence`, text === expectText);
    check(`[${locale}] ${name}: no internal text on screen`, !LEAK.test(await page.locator('body').innerText()));
    const joined = wire.join('\n');
    check(`[${locale}] ${name}: action response on the wire has no internal text (${wire.length} resp, ${joined.length} B)`, wire.length > 0 && !LEAK.test(joined), (joined.match(/"code":"[A-Z_]+"/g) ?? []).join(','));
    await page.screenshot({ path: `${SP}/shots/${shot}.png`, fullPage: true });
  } finally { psql(f.remove); }
}
