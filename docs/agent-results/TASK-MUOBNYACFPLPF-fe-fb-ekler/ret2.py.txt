import re,os
acts=[]
for dp,_,fs in os.walk('src'):
    for f in fs:
        if re.search(r'\.tsx?$',f) and '.test.' not in f:
            p=os.path.join(dp,f); s=open(p).read()
            if re.match(r"\s*(//[^\n]*\n\s*|/\*[\s\S]*?\*/\s*)*['\"]use server['\"]",s): acts.append(p)
for p in sorted(acts):
    s=open(p).read()
    imported=set()
    for m in re.finditer(r"import\s+(?!type)\{([^}]*)\}\s+from\s+['\"](@/|\.)[^'\"]*['\"]",s):
        for n in re.findall(r"(?:^|,)\s*(?:type\s+)?([A-Za-z_$][\w$]*)(?:\s+as\s+([\w$]+))?",m.group(1)): imported.add(n[1] or n[0])
    for m in re.finditer(r"(?:const|let)\s+([A-Za-z_$][\w$]*)\s*=\s*await\s+([A-Za-z_$][\w$]*)\(",s):
        v,fn=m.group(1),m.group(2)
        if fn not in imported or fn in ('requireTenantAction','requirePlatformAdmin','auth','getTranslations','withTenant'): continue
        rest=s[m.end():m.end()+4000]
        # return v; / return { ...v } / key: v, / v  as whole in return object
        if re.search(r"return\s+"+v+r"\s*;|\.\.\."+v+r"\b|return\s*\{[^}]*\b"+v+r"\b(?!\.)",rest):
            ln=s[:m.start()].count('\n')+1
            print(f"{p.replace('src/app/[locale]/','')}:{ln}: {v} = await {fn}(...) returned whole")
