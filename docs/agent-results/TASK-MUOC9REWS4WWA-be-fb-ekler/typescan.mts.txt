// adı error/reason/message/detail/cause/stack olan ve tipi SERBEST string (literal birliği değil) olan alanları listeler.
import ts from 'typescript';
import { readFileSync } from 'node:fs';
const cfg = ts.getParsedCommandLineOfConfigFile('tsconfig.json', {}, { ...ts.sys, onUnRecoverableConfigFileDiagnostic: () => {} })!;
const program = ts.createProgram(cfg.fileNames, cfg.options);
const tc = program.getTypeChecker();
const NAME = /^(error|errors|reason|message|msg|detail|details|cause|stack|err|description)$/i;
const isTopUseServer = (sf: ts.SourceFile) => sf.statements.some((s) => ts.isExpressionStatement(s) && ts.isStringLiteral(s.expression) && s.expression.text === 'use server');
function freeStrings(t: ts.Type, path: string, seen: Set<ts.Type>, out: string[], depth = 0) {
  if (depth > 6 || seen.has(t)) return; seen.add(t);
  if (t.isUnion()) { for (const u of t.types) freeStrings(u, path, seen, out, depth); return; }
  const p = tc.getPromisedTypeOfPromise(t); if (p) return freeStrings(p, path, seen, out, depth);
  if (tc.isArrayType(t)) { const [e] = tc.getTypeArguments(t as ts.TypeReference); if (e) freeStrings(e, path + '[]', seen, out, depth + 1); return; }
  if (!(t.flags & ts.TypeFlags.Object)) return;
  for (const prop of tc.getPropertiesOfType(t)) {
    const decl = prop.valueDeclaration ?? prop.declarations?.[0]; if (!decl) continue;
    const pt = tc.getTypeOfSymbolAtLocation(prop, decl);
    const q = `${path}.${prop.name}`;
    const parts = pt.isUnion() ? pt.types : [pt];
    const free = parts.some((x) => x.flags & ts.TypeFlags.String);  // string (literal değil)
    if (NAME.test(prop.name) && free) out.push(`${q}: ${tc.typeToString(pt)}`);
    if (!(pt.flags & (ts.TypeFlags.StringLike | ts.TypeFlags.NumberLike | ts.TypeFlags.BooleanLike))) freeStrings(pt, q, seen, out, depth + 1);
  }
}
let files = 0, fns = 0; const hits: string[] = [];
for (const sf of program.getSourceFiles()) {
  if (!sf.fileName.includes('/src/') || sf.isDeclarationFile || !isTopUseServer(sf)) continue;
  files++;
  const mod = tc.getSymbolAtLocation(sf); if (!mod) continue;
  for (const ex of tc.getExportsOfModule(mod)) {
    const sym = ex.flags & ts.SymbolFlags.Alias ? tc.getAliasedSymbol(ex) : ex;
    const d = sym.valueDeclaration; if (!d) continue;
    const sigs = tc.getTypeOfSymbolAtLocation(sym, d).getCallSignatures(); if (!sigs.length) continue;
    fns++;
    const out: string[] = [];
    freeStrings(tc.getReturnTypeOfSignature(sigs[0]), '', new Set(), out);
    for (const o of out) hits.push(`${sf.fileName.split('/src/')[1]} ${ex.name} → ret${o}`);
  }
}
console.log(`use-server dosya=${files} export fonksiyon=${fns} serbest-string hata-adlı alan=${hits.length}`);
for (const h of hits) console.log(h);
