import { afterAll, beforeAll, describe, expect, test, vi } from 'vitest';

import { prisma } from '@/lib/db';
import { requireId } from './helpers/teardown-guard';

// be-fb İNCELEME-73 attack file — NOT committed. Only auth(), the module gate and
// the LLM text are faked; role gate, withTenant, Prisma and action bodies are real.
const s = vi.hoisted(() => ({ session: null as Record<string, unknown> | null, llmCalls: 0 }));
vi.mock('@/lib/auth', () => ({ auth: vi.fn(async () => s.session) }));
vi.mock('next/cache', () => ({ revalidatePath: vi.fn(), revalidateTag: vi.fn() }));
vi.mock('@/lib/modules/access', async (orig) => ({ ...(await orig<object>()), assertModuleAccess: vi.fn(async () => {}) }));
vi.mock('@/lib/llm', async (orig) => ({
  ...(await orig<object>()),
  generateText: vi.fn(async () => {
    s.llmCalls++;
    return 'GIZLILIK SOZLESMESI (NDA) ... taraflar ...';
  }),
}));

const m10 = await import('@/modules/m10/server/actions');

const TAG = `i73-${Date.now()}`;
const fx: Record<string, string> = {};
const as = (role: string) => {
  s.session = { user: { id: `${TAG}-${role}`, name: `${TAG} ${role}`, email: `${role.toLowerCase()}@${TAG}.test` }, tenantId: fx.a, role };
};
const count = () => prisma.followupDoc.count({ where: { tenantId: requireId(fx.a, 't') } });
const row = (id: string) => prisma.followupDoc.findUnique({ where: { id }, select: { kind: true, signedAt: true } });
// A stored kind written straight into the table (a legacy row, an import, a manual fix).
const raw = (kind: string) =>
  prisma.followupDoc.create({ data: { tenantId: requireId(fx.a, 't'), opportunityId: requireId(fx.oa, 'o'), kind } });

beforeAll(async () => {
  const t = await prisma.tenant.create({ data: { name: `${TAG} a`, slug: `${TAG}-a`, country: 'TR' } });
  const ws = await prisma.workspace.create({ data: { tenantId: t.id, name: 'ws', slug: `${TAG}-ws` } });
  const opp = await prisma.opportunity.create({ data: { tenantId: t.id, workspaceId: ws.id, title: 'opp' } });
  Object.assign(fx, { a: t.id, oa: opp.id });
});
afterAll(async () => {
  if (fx.a) await prisma.tenant.delete({ where: { id: requireId(fx.a, 't') } });
});

const LONG = 'NDA' + ' '.repeat(1_000_000);
// [label, payload]
const WRITE_ATTACKS: Array<[string, unknown]> = [
  ['A1 "nda"', 'nda'],
  ['A1 "Nda"', 'Nda'],
  ['A1 "NDA "', 'NDA '],
  ['A1 " NDA"', ' NDA'],
  ['A1 "loi"', 'loi'],
  ['A1 "Commitment"', 'Commitment'],
  ['A1 "NDA+ZWSP"', 'NDA​'],
  ['homoglyph Cyrillic Н ("НDA")', 'НDA'],
  ['homoglyph Cyrillic А ("NDА")', 'NDА'],
  ['NBSP "NDA\\u00A0"', 'NDA '],
  ['full-width "ＮＤＡ"', 'ＮＤＡ'],
  ['mixed "nDa"', 'nDa'],
  ['direct-kind variant "esr_debrief"', 'esr_debrief'],
  ['direct-kind dotless ı "ESR_DEBRıEF"', 'ESR_DEBRıEF'],
  ['array ["NDA"]', ['NDA']],
  ['object {toString}', { toString: () => 'NDA' }],
  ['String object', new String('ESR_DEBRIEF')],
  ['null', null],
  ['undefined', undefined],
  ['number 0', 0],
  ['empty ""', ''],
  ['very long (1 MB)', LONG],
];

describe('generateStandardDoc: every non-exact kind is refused before the LLM, no row', () => {
  test.each(WRITE_ATTACKS)('%s', async (_label, payload) => {
    as('WRITER');
    const before = await count();
    const llmBefore = s.llmCalls;
    await expect(m10.generateStandardDoc(fx.oa, payload as never)).rejects.toThrow(/Unknown document kind/);
    expect(await count()).toBe(before);
    expect(s.llmCalls).toBe(llmBefore);
  });
  test('counterfactual: exact ESR_DEBRIEF is written and signs directly', async () => {
    as('WRITER');
    const gen = await m10.generateStandardDoc(fx.oa, 'ESR_DEBRIEF');
    expect((await row(gen.id))?.kind).toBe('ESR_DEBRIEF');
    expect((await m10.signDocument(gen.id)).success).toBe(true);
  });
  test('exact NDA is written but signDocument refuses DUAL_APPROVAL_REQUIRED', async () => {
    as('WRITER');
    const gen = await m10.generateStandardDoc(fx.oa, 'NDA');
    expect(await m10.signDocument(gen.id)).toEqual({ success: false, signedAt: null, refusal: 'DUAL_APPROVAL_REQUIRED' });
    expect((await row(gen.id))?.signedAt).toBeNull();
  });
});

// Rows planted directly in the DB: the gate must be fail-closed on every one
// that is not a direct-sign kind.
const REFUSED_ROWS = [
  'nda', 'Nda', 'NDA ', ' NDA', 'loi', 'Commitment', 'NDA​',
  'НDA', 'NDА', 'NDA ', 'ＮＤＡ', 'nDa',
  'ŃDA', 'NDA\u0000'.replace('\u0000', '⁠'), '‮NDA', '﻿NDA',
  'CONTRACT', '', ' ', 'ESR_DEBRİEF', 'ESR-DEBRIEF', 'ESR_DEBRIEF_NDA', 'AWARD PACK', LONG,
];
describe('signDocument on planted rows: fail-closed', () => {
  test.each(REFUSED_ROWS.map((k) => [JSON.stringify(k).slice(0, 40), k]))('row kind %s → DUAL_APPROVAL_REQUIRED, unchanged', async (_l, kind) => {
    const d = await raw(kind);
    as('WRITER');
    expect(await m10.signDocument(d.id)).toEqual({ success: false, signedAt: null, refusal: 'DUAL_APPROVAL_REQUIRED' });
    expect((await row(d.id))?.signedAt).toBeNull();
  });
});

// Record-only: spellings that normalise onto a DIRECT kind are signed directly
// by design (the gate folds them). Logged, not asserted as an attack.
const FOLDED_TO_DIRECT = ['esr_debrief', ' ESR_DEBRIEF ', 'ESR_DEBRıEF', 'ＥＳＲ_DEBRIEF', 'award_pack', 'AWARD_PACK​'];
describe('record: variants that fold onto a direct-sign kind', () => {
  test.each(FOLDED_TO_DIRECT.map((k) => [JSON.stringify(k), k]))('row kind %s', async (_l, kind) => {
    const d = await raw(kind);
    as('WRITER');
    const res = await m10.signDocument(d.id);
    console.log(`FOLD ${JSON.stringify(kind)} → ${JSON.stringify(res.success)}`);
    expect(res.success).toBe(true);
  });
});
