// TASK-MUO2B59ZNY7D7 live checks against :3482 (DB csfb-o2b5-pg only). Chrome via @playwright/test.
const { chromium } = require('@playwright/test');
const { execSync } = require('child_process');
const B = 'http://localhost:3482';
const DIR = process.env.SHOTS;
const q = (sql) => execSync(`docker exec -i csfb-o2b5-pg psql -U fabrika -d fabrika -tAq`, { input: sql }).toString().trim();

async function login(ctx, email) {
  const r = ctx.request;
  const { csrfToken } = await (await r.get(B + '/api/auth/csrf')).json();
  await r.post(B + '/api/auth/callback/credentials', { form: { csrfToken, email, callbackUrl: B + '/tr' }, maxRedirects: 0 });
}
const sess = async (ctx) => {
  const s = await (await ctx.request.get(B + '/api/auth/session')).json();
  return JSON.stringify({ tenantId: s?.tenantId ?? null, role: s?.role ?? null, keys: Object.keys(s || {}) });
};
async function update(ctx, data) {
  const { csrfToken } = await (await ctx.request.get(B + '/api/auth/csrf')).json();
  const res = await ctx.request.post(B + '/api/auth/session', {
    headers: { 'content-type': 'application/json' },
    data: JSON.stringify({ csrfToken, data }),
    failOnStatusCode: false,
    maxRedirects: 0,
  });
  const j = await res.json().catch(() => null);
  return `${res.status()} ${JSON.stringify({ tenantId: j?.tenantId ?? null, role: j?.role ?? null })}`;
}
async function dashboard(ctx, shot) {
  const p = await ctx.newPage();
  await p.goto(B + '/tr/default/dashboard', { waitUntil: 'load', timeout: 180000 });
  await p.waitForTimeout(3000);
  const url = p.url().replace(B, '');
  const body = (await p.innerText('body')).replace(/\s+/g, ' ');
  if (shot) await p.screenshot({ path: `${DIR}/${shot}` });
  await p.close();
  return `${url} | mentions D: ${/SILINMIS D/.test(body)} | A: ${/cfo2 A/.test(body)}`;
}

(async () => {
  const b = await chromium.launch({ channel: 'chrome' });
  const ctx = await b.newContext({ viewport: { width: 1280, height: 800 } });
  await login(ctx, 'cfo2@fabrika.test');
  console.log('[0]  signed in (A oldest, live)                  :', await sess(ctx));
  console.log('[A9] update {tenantId: cfo2-d} (member, DELETED) :', await update(ctx, { tenantId: 'cfo2-d' }), '| next GET', await sess(ctx));
  console.log('[A9b] dashboard after A9                         :', await dashboard(ctx, 'shot-A9-after-update.png'));
  console.log('[A8] update {tenantId: cfo2-b} (member, live)    :', await update(ctx, { tenantId: 'cfo2-b' }), '| next GET', await sess(ctx));
  console.log('[A10] update {tenantId: cfo2-a} (older, live)    :', await update(ctx, { tenantId: 'cfo2-a' }), '| next GET', await sess(ctx));

  // Bound tenant soft-deleted mid-session, then restored (deletedAt cleared).
  q(`update "Tenant" set "deletedAt" = now() where id = 'cfo2-a'`);
  console.log('[R1] A deleted, plain GET ×3                     :', await sess(ctx), await sess(ctx), await sess(ctx));
  console.log('[R2] dashboard while A deleted                   :', await dashboard(ctx, 'shot-R2-bound-deleted.png'));
  console.log('[R3] update {tenantId: cfo2-a} while deleted     :', await update(ctx, { tenantId: 'cfo2-a' }));
  q(`update "Tenant" set "deletedAt" = null where id = 'cfo2-a'`);
  console.log('[R4] A restored, plain GET (same cookie)         :', await sess(ctx));
  console.log('[R5] dashboard after restore                     :', await dashboard(ctx, 'shot-R5-restored.png'));

  // Sign-in where the OLDEST membership is in a deleted tenant.
  const ctx2 = await b.newContext();
  await login(ctx2, 'cfo2-w@fabrika.test');
  console.log('[P1] w signs in (D oldest+deleted, B live)       :', await sess(ctx2));
  console.log('[P2] w update {tenantId: cfo2-d}                 :', await update(ctx2, { tenantId: 'cfo2-d' }), '| next GET', await sess(ctx2));
  console.log('[DB] memberships u:', q(`select string_agg("tenantId"||':'||role, ',' order by "createdAt") from "Membership" where "userId"='cfo2-u'`));
  await b.close();
})().catch((e) => { console.error(e); process.exit(1); });
