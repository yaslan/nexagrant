import { test, expect } from '@playwright/test';
import { writeFileSync } from 'node:fs';
import { prisma } from '../../src/lib/db';
import { evidenceTenant } from './fixtures/evidence-tenant';
import { expectHydrated } from './fixtures/hydration';
import { signInAs } from './fixtures/session';

const OUT = process.env.R2_PROOF_OUT!;
const tenant = evidenceTenant('r2proof');

test.beforeAll(async () => { await tenant.setup(); });
test.afterAll(async () => { await tenant.teardown(); await prisma.$disconnect(); });

test('production without R2: upload refused early, no mock-upload request', async ({ page }) => {
  const requests: string[] = [];
  page.on('request', (r) => requests.push(`${r.method()} ${r.url()}`));
  const dialogs: string[] = [];
  page.on('dialog', async (d) => { dialogs.push(d.message()); await d.dismiss(); });

  await signInAs(page, { id: 'ev-r2proof-e2e-user', email: tenant.email, tenantId: tenant.tenantId, role: 'TENANT_OWNER' });
  await page.goto(tenant.evidenceUrl);
  await expectHydrated(page.getByRole('button', { name: /Belge Yükle|Upload/ }).first());

  await page.getByRole('button', { name: /Yeni Belge Yükle/ }).first().click();
  await page.locator('input[type="file"]').setInputFiles({ name: 'r2-proof.pdf', mimeType: 'application/pdf', buffer: Buffer.from('%PDF-1.4\n%%EOF\n') });
  await page.getByRole('button', { name: /Cloudflare R2/ }).click();

  const banner = page.getByTestId('evidence-upload-error');
  await expect(banner).toBeVisible({ timeout: 15_000 });
  await expect(banner).toContainText('Belge yükleme bu kurulumda henüz açık değil');
  await page.screenshot({ path: `${OUT}/prod-no-r2-upload-refused.png`, fullPage: false });

  const mock = requests.filter((r) => r.includes('mock-upload'));
  const puts = requests.filter((r) => r.startsWith('PUT '));
  writeFileSync(`${OUT}/network.txt`, [
    `banner: ${await banner.textContent()}`,
    `dialogs (native alert): ${JSON.stringify(dialogs)}`,
    `requests total: ${requests.length}`,
    `requests containing "mock-upload": ${mock.length}`,
    `PUT requests: ${puts.length}`,
    'POST requests (server actions):',
    ...requests.filter((r) => r.startsWith('POST ')),
  ].join('\n') + '\n');
  expect(mock).toHaveLength(0);
  expect(puts).toHaveLength(0);
  expect(dialogs).toHaveLength(0);
  const rows = await prisma.evidence.count({ where: { tenantId: tenant.tenantId } });
  expect(rows).toBe(0);
});
