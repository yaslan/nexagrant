// crw-fb İNCELEME-94 probe — NOT committed. Real Prisma + real repository + real requirePlatformAdmin
// against the isolated DB crwfb94-pg (:5494). Only auth() and revalidatePath are stubbed.
import { describe, it, expect, vi, beforeEach, afterAll } from 'vitest';
import { execFileSync } from 'node:child_process';
import { appendFileSync } from "node:fs";
const OUT = process.env.CRW94_OUT ?? "/dev/null";
import tr from '../../messages/tr.json';
import en from '../../messages/en.json';

const s = vi.hoisted(() => ({ session: null as unknown }));
vi.mock('next/cache', () => ({ revalidatePath: vi.fn() }));
vi.mock('@/lib/auth', () => ({ auth: vi.fn(async () => s.session) }));

import { logger } from '@/lib/logger';
import * as A from '@/app/[locale]/admin/billing/plans/actions';
import { PLAN_FAILURE_CODES } from '@/app/[locale]/admin/billing/plans/failure-codes';

const psql = (sql: string) =>
  execFileSync('docker', ['exec', 'crwfb94-pg', 'psql', '-U', 'fabrika', '-d', 'fabrika', '-v', 'ON_ERROR_STOP=1', '-tAc', sql]).toString().trim();

const ADMIN = { user: { id: 'demo-user-alice', isPlatformAdmin: true } };
const LEAK = /P\d{4}|\d{5}|prisma|invocation|constraint|Unique|relation|does not exist|crw94|"Plan/i;
const errs = (m: unknown) => (m as { adminPlans: { errors: Record<string, string> } }).adminPlans.errors;

const CALLS: Record<string, () => Promise<unknown>> = {
  createDraftAction: () => A.createDraftAction('ultra'),
  saveDraftAction: () => A.saveDraftAction('pv_ultra_1', {} as never, []),
  discardDraftAction: () => A.discardDraftAction('pv_ultra_1'),
  previewImpactAction: () => A.previewImpactAction('pv_ultra_1'),
  publishVersionAction: () => A.publishVersionAction('pv_ultra_1', 0),
  setVisibilityAction: () => A.setVisibilityAction('ultra', 'public'),
  archivePlanAction: () => A.archivePlanAction('ultra'),
  deletePlanAction: () => A.deletePlanAction('ultra'),
  assignPlanAction: () => A.assignPlanAction('public-team', 'demo-tenant', ''),
  unassignPlanAction: () => A.unassignPlanAction('public-team', 'demo-tenant'),
  grantPilotAction: () => A.grantPilotAction('demo-tenant', false),
};

function show(tag: string, name: string, res: unknown) {
  const r = res as { ok: boolean; code?: string };
  const code = r.ok ? '(ok)' : r.code!;
  const trS = r.ok ? '' : errs(tr)[code];
  const enS = r.ok ? '' : errs(en)[code];
  const logged = (vi.mocked(logger.error).mock.calls.at(-1)?.[1] ?? null) as { code?: unknown; constructor?: { name?: string } } | null;
  appendFileSync(OUT, `${tag} ${name}: ${JSON.stringify(res)} | tr="${trS}" | en="${enS}" | log.code=${logged ? String((logged as { code?: unknown }).code) : '-'} log.class=${logged?.constructor?.name ?? "-"}\n`);
  return { code, trS, enS, logged };
}

beforeEach(() => {
  vi.restoreAllMocks();
  vi.spyOn(logger, 'error').mockImplementation(() => {});
  vi.spyOn(logger, 'warn').mockImplementation(() => {});
  s.session = ADMIN;
});

describe('crw94 real-DB probe', () => {
  it('S1 non-admin session and no session: PLATFORM_ADMIN_REQUIRED on all 11, DB untouched', async () => {
    const before = psql(`select md5((select string_agg(t::text, ',' order by t::text) from "Plan" t) || coalesce((select string_agg(t::text, ',' order by t::text) from "PlanAssignment" t),'') || (select string_agg(t::text, ',' order by t::text) from "PlanVersion" t) || coalesce((select string_agg(t::text, ',' order by t::text) from "Subscription" t),''))`);
    for (const sess of [{ user: { id: 'demo-user-bob' } }, { user: { id: 'demo-user-bob', role: 'ADMIN' }, role: 'ADMIN' }, null]) {
      s.session = sess;
      for (const [name, call] of Object.entries(CALLS)) {
        const res = await call();
        const { trS, enS } = show('S1', name, res);
        expect(res).toStrictEqual({ ok: false, code: 'PLATFORM_ADMIN_REQUIRED' });
        expect(trS && enS).toBeTruthy();
      }
    }
    expect(logger.error).not.toHaveBeenCalled();
    expect(psql(`select md5((select string_agg(t::text, ',' order by t::text) from "Plan" t) || coalesce((select string_agg(t::text, ',' order by t::text) from "PlanAssignment" t),'') || (select string_agg(t::text, ',' order by t::text) from "PlanVersion" t) || coalesce((select string_agg(t::text, ',' order by t::text) from "Subscription" t),''))`)).toBe(before);
  });

  it('S2 tables renamed (P2021-class): all 11 answer UNKNOWN, raw code only in log', async () => {
    psql(`ALTER TABLE "PlanAssignment" RENAME TO "crw94_pa"; ALTER TABLE "PlanVersion" RENAME TO "crw94_pv"; ALTER TABLE "Plan" RENAME TO "crw94_p";`);
    try {
      for (const [name, call] of Object.entries(CALLS)) {
        vi.mocked(logger.error).mockClear();
        const res = await call();
        const { logged } = show('S2', name, res);
        expect(res).toStrictEqual({ ok: false, code: 'UNKNOWN' });
        expect(JSON.stringify(res)).not.toMatch(LEAK);
        expect(logger.error).toHaveBeenCalledTimes(1);
        expect(logged).not.toBeNull();
      }
    } finally {
      psql(`ALTER TABLE "crw94_p" RENAME TO "Plan"; ALTER TABLE "crw94_pv" RENAME TO "PlanVersion"; ALTER TABLE "crw94_pa" RENAME TO "PlanAssignment";`);
    }
  });

  it('S3 trigger RAISE with internal text on every write (P0001 → P2010/P2039-class)', async () => {
    const tables = ['Plan', 'PlanVersion', 'PlanFeature', 'PlanAssignment', 'Subscription', 'AuditEvent'];
    psql(`CREATE OR REPLACE FUNCTION crw94_boom() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN RAISE EXCEPTION 'crw94-internal tablo=% secret=hunter2', TG_TABLE_NAME; END $$;` +
      tables.map((t) => `CREATE TRIGGER crw94_boom BEFORE INSERT OR UPDATE OR DELETE ON "${t}" FOR EACH ROW EXECUTE FUNCTION crw94_boom();`).join(''));
    try {
      for (const [name, call] of Object.entries(CALLS)) {
        vi.mocked(logger.error).mockClear();
        const res = await call();
        show('S3', name, res);
        expect((res as { ok: boolean }).ok === true || PLAN_FAILURE_CODES.includes((res as { code: never }).code)).toBe(true);
        expect(JSON.stringify(res)).not.toMatch(/hunter2|crw94|P\d{4}/);
        expect(Object.keys(res as object).sort()).toEqual((res as { ok: boolean }).ok ? expect.any(Array) : ['code', 'ok']);
      }
    } finally {
      psql(tables.map((t) => `DROP TRIGGER IF EXISTS crw94_boom ON "${t}";`).join('') + 'DROP FUNCTION crw94_boom();');
    }
  });

  it('S4 FK / unique / validation on real data', async () => {
    const cases: Array<[string, () => Promise<unknown>]> = [
      ['assign→missing tenant (FK)', () => A.assignPlanAction('public-team', 'crw94-no-such-tenant', '')],
      ['assign→missing plan (FK)', () => A.assignPlanAction('crw94-no-such-plan', 'demo-tenant', '')],
      ['createDraft→missing plan', () => A.createDraftAction('crw94-no-such-plan')],
      ['createDraft twice (unique?) #1', () => A.createDraftAction('pro')],
      ['createDraft twice (unique?) #2', () => A.createDraftAction('pro')],
      ['saveDraft garbage body (Flight)', () => A.saveDraftAction('pv_pro_1', { priceMinor: 'x' } as never, [{ bogus: 1 } as never])],
      ['publish published (domain)', () => A.publishVersionAction('pv_ultra_1', 0)],
      ['setVisibility bogus enum', () => A.setVisibilityAction('ultra', 'crw94' as never)],
      ['deletePlan undefined id', () => A.deletePlanAction(undefined as never)],
      ['grantPilot missing tenant', () => A.grantPilotAction('crw94-no-such-tenant', false)],
    ];
    for (const [label, call] of cases) {
      vi.mocked(logger.error).mockClear();
      const res = await call();
      show('S4', label, res);
      if (!(res as { ok: boolean }).ok) {
        expect(PLAN_FAILURE_CODES).toContain((res as { code: string }).code);
        expect(Object.keys(res as object).sort()).toEqual(['code', 'ok']);
      }
      expect(JSON.stringify(res)).not.toMatch(/P\d{4}|crw94|invocation|constraint/);
    }
  });

  afterAll(() => {
    // clean any draft S4 opened on 'pro'
    psql(`DELETE FROM "PlanFeature" WHERE "planVersionId" IN (select id from "PlanVersion" where status='draft'); DELETE FROM "PlanVersion" WHERE status='draft';`);
  });
});
