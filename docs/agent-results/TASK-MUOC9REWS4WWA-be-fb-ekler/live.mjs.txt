const LEAK = /fetch failed|ECONNREFUSED|127\.0\.0\.1:9|10\.255\.255\.1|TimeoutError|aborted due to timeout|Failed to download|upstream down|cordis\.europa\.eu\/data|invalid signature|FILE_ENDED|end of central|Required CORDIS CSV|skipped records over|SIRCSVSATIRI|CordisOrganization|CordisImportRun|does not exist|Transaction already|expired transaction|\/srv\/secret|tmp_cordis|Invalid `prisma/;
const leakOf = (t, stored) => {
  const m = t.match(LEAK); if (m) return m[0];
  if (stored && stored.length >= 16 && t.includes(stored.slice(0, 24))) return 'saklı metnin ilk 24 karakteri: ' + stored.slice(0, 24);
  return null;
};
async function action(name, args) {
  const res = await fetch(`${B}/tr/admin/cordis`, { method: 'POST', body: JSON.stringify(args), headers: {
    'Next-Action': idOf(name), Accept: 'text/x-component', 'Content-Type': 'text/plain;charset=UTF-8', Origin: B, Cookie: `authjs.session-token=${admin}` } });
  const text = await res.text();
  return { status: res.status, text, line: ((text.match(/^1:(.*)$/m) || [])[1] ?? text.slice(0, 160)).slice(0, 160) };
}

const lastAlert = () => psql(`select left(coalesce(meta->>'error','-'),100) from "SystemAlert" where source='k2_cordis_import' and "resolvedAt" is null order by "createdAt" desc limit 1`) || '(yeni alarm satırı yok)';
