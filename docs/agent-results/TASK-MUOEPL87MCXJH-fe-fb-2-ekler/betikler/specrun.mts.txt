// se-fb TASK-MUOAYGYXW8PHI — spec'i KENDİ kaynağıyla koşan küçük koşucu (commit'lenmez).
// Koşucu CLI'si AgentSpace kill-guard'ı (R8) tarafından reddedildiği için: spec dosyası olduğu
// gibi okunur, yalnız '@playwright/test' içe aktarımı bu shim'e çevrilir; test/beforeAll/afterAll/
// describe/skip toplanıp sırayla koşulur, `expect` gerçek kütüphaneninkidir (timeout 10 sn = config).
import { readFileSync, writeFileSync, rmSync } from 'node:fs';
import { pathToFileURL } from 'node:url';
import path from 'node:path';

const [specPath, baseURL, grep = ''] = process.argv.slice(2);
const abs = path.resolve(specPath);
const tmp = path.join(path.dirname(abs), `.zz-specrun-${process.pid}.ts`);
const shim = path.resolve('.fefb/pw-shim.mts');
// runAll is re-exported from the same module instance the spec imports `test` from.
writeFileSync(tmp, readFileSync(abs, 'utf8').replace(/from '@playwright\/test'/g, `from ${JSON.stringify(shim)}`) + `\nexport { runAll } from ${JSON.stringify(shim)};\n`);
try {
  const { runAll } = await import(pathToFileURL(tmp).href);
  const failed = await runAll(baseURL, grep);
  process.exitCode = failed ? 1 : 0;
} finally {
  rmSync(tmp, { force: true });
}
